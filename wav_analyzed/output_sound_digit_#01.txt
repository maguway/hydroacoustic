        constexpr float PI_F = 3.14159265358979323846f;
        constexpr float TWO_PI_F = 6.28318530717958647692f;

        // ============================================================
        //  ДЕТАЛИ — параметры (универсальные)
        // ============================================================

        constexpr int FRAME_HOP_MS = 5; // для деталей — мелкий шаг
        constexpr int FFT_WIN = 1024;

        constexpr float F0_MIN_HZ = 100.0f;
        constexpr float F0_MAX_HZ = 8000.0f;

        constexpr float HARMONIC_THR_TONE = 0.30f;
        constexpr float SILENCE_OFFSET_DB = 12.0f;

        constexpr double MIN_EVENT_MS = 100.0;

        constexpr float AUTOCORR_MIN = 0.30f;
        constexpr int MEDIAN_WIN = 5;

        // Порог, с которого печатаем детали.
        constexpr double DETAIL_MIN_DUR_MS = 100.0;

        // Разбиение длинных TONE (как в events).
        constexpr float SUBEVENT_F0_JUMP = 0.25f;
        constexpr double MIN_SUBEVENT_MS = 80.0;

        // Порог STR.
        constexpr float STR_FREQ_FRAC = 0.60f;

        // ============================================================

        struct Frame
        {
            double tMs;
            float levelDb;
            float f0Hz;
            float harmonic;
            float highFrac;
            const char* type;
        };

        struct Harm
        {
            float freq;
            float relDb; // относительно основной
        };

        std::vector<std::pair<int, float>>
        findPeaks(const std::vector<float>& mag, float maxMag, float floorDb)
        {
            std::vector<std::pair<int, float>> peaks;
            const int half = int(mag.size());
            for (int k = 2; k < half - 1; ++k)
            {
                if (mag[k] > mag[k - 1] && mag[k] > mag[k + 1])
                {
                    const float db = 20.0f * std::log10(mag[k] / maxMag + 1e-9f);
                    if (db < floorDb) continue;
                    peaks.push_back({k, mag[k]});
                }
            }
            return peaks;
        }

        float estimateF0Autocorr(const std::vector<float>& samples,
                                 int sr, float fMin, float fMax,
                                 float& outCorr)
        {
            outCorr = 0.0f;
            const int n = int(samples.size());
            if (n < 64) return 0.0f;

            double mean = 0.0;
            for (float v : samples) mean += v;
            mean /= double(n);

            std::vector<float> x(n);
            for (int i = 0; i < n; ++i) x[i] = float(samples[i] - mean);

            double energy = 0.0;
            for (int i = 0; i < n; ++i) energy += double(x[i]) * x[i];
            if (energy < 1e-12) return 0.0f;

            const int lagMin = std::max(2, int(float(sr) / fMax));
            const int lagMax = std::min(n / 2, int(float(sr) / fMin));
            if (lagMax <= lagMin) return 0.0f;

            float bestLag = 0.0f, bestCorr = 0.0f;
            for (int lag = lagMin; lag <= lagMax; ++lag)
            {
                double sum = 0.0;
                for (int i = 0; i + lag < n; ++i)
                    sum += double(x[i]) * x[i + lag];
                const float corr = float(sum / energy);
                if (corr > bestCorr && corr > AUTOCORR_MIN)
                {
                    bestCorr = corr;
                    bestLag = float(lag);
                }
            }
            if (bestLag <= 0.0f) return 0.0f;
            outCorr = bestCorr;
            return float(sr) / bestLag;
        }

        float estimateHarmonicity(const std::vector<float>& mag,
                                  float maxMag, int sr, int fftSize,
                                  float fMin, float fMax)
        {
            auto peaks = findPeaks(mag, maxMag, -40.0f);
            if (peaks.size() < 2) return 0.0f;
            const float binHz = float(sr) / float(fftSize);
            int totalChecked = 0, harmonicCount = 0;
            for (const auto& p : peaks)
            {
                const float f1 = float(p.first) * binHz;
                if (f1 < fMin || f1 > fMax) continue;
                ++totalChecked;
                bool has2 = false, has3 = false;
                for (const auto& q : peaks)
                {
                    const float f2 = float(q.first) * binHz;
                    const float r2 = f2 / f1;
                    if (std::fabs(r2 - 2.0f) < 0.10f) has2 = true;
                    if (std::fabs(r2 - 3.0f) < 0.10f) has3 = true;
                }
                if (has2 || has3) ++harmonicCount;
            }
            if (totalChecked == 0) return 0.0f;
            return float(harmonicCount) / float(totalChecked);
        }

        float computeHighFrac(const std::vector<float>& mag, int sr, int fftSize,
                              float boundaryHz)
        {
            const int half = int(mag.size());
            const float binHz = float(sr) / float(fftSize);
            const int kb = std::max(1, int(boundaryHz / binHz));
            double eLow = 0.0, eHigh = 0.0;
            for (int k = 2; k < half; ++k)
            {
                const double e = double(mag[k]) * mag[k];
                if (k < kb) eLow += e;
                else eHigh += e;
            }
            const double eTot = eLow + eHigh;
            if (eTot < 1e-20) return 0.0f;
            return float(eHigh / eTot);
        }

        const char* classify(float levelDb, float silenceThr,
                             float f0, float harmonic, float highFrac,
                             float fMax)
        {
            if (levelDb < silenceThr) return "SILENCE";
            if (harmonic >= HARMONIC_THR_TONE)
            {
                if (f0 > fMax * STR_FREQ_FRAC) return "STR";
                if (f0 == 0.0f && highFrac > 0.75f) return "STR";
                return "TONE";
            }
            if (highFrac > 0.75f) return "STR";
            return "NOISE";
        }

        void smoothFrames(std::vector<Frame>& frames)
        {
            const int n = int(frames.size());
            if (n < 4) return;
            for (int pass = 0; pass < 2; ++pass)
            {
                for (int i = 1; i < n - 1; ++i)
                {
                    if (frames[i].type != frames[i - 1].type
                        && frames[i].type != frames[i + 1].type)
                    {
                        frames[i].type = frames[i - 1].type;
                    }
                }
            }
        }

        void medianFilterF0(std::vector<Frame>& frames)
        {
            const int n = int(frames.size());
            if (n < MEDIAN_WIN) return;
            std::vector<float> src(n);
            for (int i = 0; i < n; ++i) src[i] = frames[i].f0Hz;
            const int half = MEDIAN_WIN / 2;
            std::vector<float> win;
            win.reserve(MEDIAN_WIN);
            for (int i = 0; i < n; ++i)
            {
                win.clear();
                for (int j = -half; j <= half; ++j)
                {
                    const int k = i + j;
                    if (k < 0 || k >= n) continue;
                    if (src[k] <= 0.0f) continue;
                    win.push_back(src[k]);
                }
                if (win.empty())
                {
                    frames[i].f0Hz = 0.0f;
                    continue;
                }
                std::sort(win.begin(), win.end());
                frames[i].f0Hz = win[win.size() / 2];
            }
        }

        struct Event
        {
            double startMs, endMs;
            const char* type;
            float f0Start, f0End;
            float levelDbMean;
            std::vector<Frame> frames;
        };

        std::vector<Event> splitTone(const Event& e)
        {
            std::vector<Event> subs;
            if (e.frames.size() < 3)
            {
                subs.push_back(e);
                return subs;
            }

            size_t start = 0;
            for (size_t i = 1; i < e.frames.size(); ++i)
            {
                const float f0Prev = e.frames[i - 1].f0Hz;
                const float f0Curr = e.frames[i].f0Hz;
                if (f0Prev <= 0.0f || f0Curr <= 0.0f) continue;

                const float jump = std::fabs(f0Curr - f0Prev)
                    / std::max(50.0f, f0Prev);
                const float totalDur = e.frames[i].tMs - e.frames[start].tMs;

                if (jump > SUBEVENT_F0_JUMP && totalDur > MIN_SUBEVENT_MS)
                {
                    Event sub;
                    sub.startMs = e.frames[start].tMs;
                    sub.endMs = e.frames[i - 1].tMs;
                    sub.type = e.type;
                    sub.f0Start = e.frames[start].f0Hz;
                    sub.f0End = e.frames[i - 1].f0Hz;
                    float sumL = 0.0f;
                    for (size_t k = start; k < i; ++k) sumL += e.frames[k].levelDb;
                    sub.levelDbMean = sumL / float(i - start);
                    sub.frames.assign(e.frames.begin() + start,
                                      e.frames.begin() + i);
                    subs.push_back(sub);
                    start = i;
                }
            }
            if (start < e.frames.size())
            {
                Event sub;
                sub.startMs = e.frames[start].tMs;
                sub.endMs = e.frames.back().tMs;
                sub.type = e.type;
                sub.f0Start = e.frames[start].f0Hz;
                sub.f0End = e.frames.back().f0Hz;
                float sumL = 0.0f;
                for (size_t k = start; k < e.frames.size(); ++k)
                    sumL += e.frames[k].levelDb;
                sub.levelDbMean = sumL / float(e.frames.size() - start);
                sub.frames.assign(e.frames.begin() + start, e.frames.end());
                subs.push_back(sub);
            }

            std::vector<Event> kept;
            for (auto& s : subs)
            {
                const double dur = s.endMs - s.startMs;
                if (dur < MIN_SUBEVENT_MS && !kept.empty())
                {
                    kept.back().endMs = s.endMs;
                    kept.back().f0End = s.f0End;
                    for (const auto& fr : s.frames)
                        kept.back().frames.push_back(fr);
                    continue;
                }
                kept.push_back(s);
            }
            return kept;
        }

        // ============================================================
        //  Для кадра и его F0 — найти топ-3 гармоники (2×, 3×, 4×).
        //  Возвращает вектор пар (частота, дБ относительно основной).
        // ============================================================
        std::vector<Harm> findHarmonics(const std::vector<float>& mag,
                                        int sr, int fftSize,
                                        float f0)
        {
            std::vector<Harm> result;
            if (f0 <= 0.0f) return result;

            const float binHz = float(sr) / float(fftSize);
            const int half = int(mag.size());

            // Уровень основной: ближайший бин к f0.
            const int k0 = std::max(1, int(f0 / binHz + 0.5f));
            float mag0 = 0.0f;
            for (int k = std::max(1, k0 - 2); k <= std::min(half - 1, k0 + 2); ++k)
                if (mag[k] > mag0) mag0 = mag[k];
            if (mag0 < 1e-9f) return result;

            // Ищем пики около 2×, 3×, 4×.
            for (int h = 2; h <= 4; ++h)
            {
                const float fH = f0 * float(h);
                if (fH >= float(sr) * 0.45f) break;
                const int kH = std::max(1, int(fH / binHz + 0.5f));

                // Локальный максимум в окне ±3 бина.
                int kBest = -1;
                float magBest = 0.0f;
                for (int k = std::max(1, kH - 3);
                     k <= std::min(half - 1, kH + 3); ++k)
                {
                    if (mag[k] > magBest)
                    {
                        magBest = mag[k];
                        kBest = k;
                    }
                }
                if (kBest < 0 || magBest < 1e-9f) continue;

                Harm hm;
                hm.freq = float(kBest) * binHz;
                hm.relDb = 20.0f * std::log10(magBest / mag0 + 1e-9f);
                if (hm.relDb < -40.0f) continue; // слишком тихо
                result.push_back(hm);
            }
            return result;
        }
    }

    void writeDetails(std::ostream& os, const WavData& wav)
    {
        const int sr = wav.sampleRate;
        const int64_t N = int64_t(wav.samples.size());
        const int win = FFT_WIN;
        const int hop = std::max(1, sr * FRAME_HOP_MS / 1000);

        if (N < win || sr <= 0)
        {
            os << "# file too short or sr=0\n";
            return;
        }

        const double totalDurMs = double(N) / double(sr) * 1000.0;
        const float fMin = F0_MIN_HZ;
        const float fMax = std::min(F0_MAX_HZ, float(sr) * 0.45f);
