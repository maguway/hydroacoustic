#pragma once

// ============================================================
//  Математические константы.
//
//  M_PI и M_TWO_PI в стандартном C++ не определены, а в GCC/MinGW
//  доступны только как POSIX-расширение (и то не всегда).
//  Поэтому определяем явно, чтобы не зависеть от платформы.
//
//  В коде используем либо макросы M_PI / M_TWO_PI (для совместимости
//  с C-кодом, если придётся), либо constexpr PI_F / TWO_PI_F
//  (для чистого C++).
// ============================================================

#ifndef M_PI
#define M_PI 3.14159265358979323846
#endif

#ifndef M_TWO_PI
#define M_TWO_PI 6.28318530717958647692
#endif

namespace hydro {

    // float-версии
    constexpr float PI_F     = 3.14159265358979323846f;
    constexpr float TWO_PI_F = 6.28318530717958647692f;

    // double-версии
    constexpr double PI_D     = 3.14159265358979323846;
    constexpr double TWO_PI_D = 6.28318530717958647692;

    // Углы
    inline float  deg2rad(float deg)  { return deg * PI_F / 180.0f; }
    inline float  rad2deg(float rad)  { return rad * 180.0f / PI_F; }
    inline double deg2rad(double deg) { return deg * PI_D / 180.0; }
    inline double rad2deg(double rad) { return rad * 180.0 / PI_D; }

} // namespace hydro

// ============================================================
//  State Variable Filter (Chamberlin).
//
//  Устойчивее biquad на краях диапазона, даёт одновременно
//  LP / BP / HP выходы, прост в реализации, легко считается.
//
//  Нормированная частота:
//      f = 2 * sin(pi * fc / fs)
//  Затухание:
//      q = 1 / Q
//
//  Коэффициенты задаются через set(). Один экземпляр = один
//  фильтр, хранит своё состояние (lp, bp).
//
//  Используется:
//    - в ocean_noise.cpp — для 4 полос волны и шума приёмника;
//    - в source_model.cpp — для полосовой окраски кавитации
//      и механического шума.
//
//  Границы:
//    fc ∈ [5 Гц, 0.45 * fs]     (иначе клампится)
//    Q  ∈ [0.5, 4.0]            (иначе клампится)
//
//  Защита от разбега: если bp или lp уходят за пределы float
//  или становятся NaN/Inf — состояние сбрасывается в 0.
// ============================================================

namespace hydro
{
    struct Svf
    {
        float f  = 0.1f;    // нормированная частота
        float q  = 1.0f;    // 1 / Q
        float lp = 0.0f;    // состояние LP-выхода
        float bp = 0.0f;    // состояние BP-выхода

        // Настройка фильтра под fs, fc и Q.
        void set(float fs, float fc, float Q)
        {
            if (fc < 5.0f)          fc = 5.0f;
            if (fc > fs * 0.45f)    fc = fs * 0.45f;
            if (Q  < 0.5f)          Q  = 0.5f;
            if (Q  > 4.0f)          Q  = 4.0f;

            f = 2.0f * std::sin(PI_F * fc / fs);
            if (f > 1.0f) f = 1.0f;
            q = 1.0f / Q;
        }

        // Прогон одного отсчёта. Возвращает BP-выход.
        // LP и BP обновляются внутри.
        float processBP(float x)
        {
            const float hp = x - lp - q * bp;
            bp += f * hp;
            lp += f * bp;

            // Защита от разбега.
            if (!std::isfinite(bp) || std::fabs(bp) > 1e6f)
            {
                bp = 0.0f;
                lp = 0.0f;
            }
            if (!std::isfinite(lp) || std::fabs(lp) > 1e6f)
            {
                lp = 0.0f;
            }

            return bp;
        }

        // Сброс состояния (для повторного использования).
        void reset()
        {
            lp = 0.0f;
            bp = 0.0f;
        }
    };

} // namespace hydro

#include "uca_geometry.h"
#include "math_constants.h"
#include <cmath>

namespace hydro {

    UcaGeometry::UcaGeometry(int numElements, float radiusMeters)
        : n_(numElements), R_(radiusMeters), phi_(numElements)
    {
        for (int i = 0; i < n_; ++i) {
            phi_[i] = TWO_PI_F * i / n_;
        }
    }

    float UcaGeometry::elementAngle(int i) const {
        return phi_[i];
    }

    void UcaGeometry::elementPosition(int i, float& x, float& y) const {
        x = R_ * std::cos(phi_[i]);
        y = R_ * std::sin(phi_[i]);
    }

    std::vector<float> UcaGeometry::delays(float thetaDeg, float c) const {
        std::vector<float> tau(n_);
        const float theta = deg2rad(thetaDeg);
        const float inv_c = 1.0f / c;
        for (int i = 0; i < n_; ++i) {
            tau[i] = R_ * std::cos(theta - phi_[i]) * inv_c;
        }
        return tau;
    }

    std::vector<float> UcaGeometry::arrivalOffsets(float thetaDeg, float c) const {
        std::vector<float> dt(n_);
        const float theta = deg2rad(thetaDeg);
        const float inv_c = 1.0f / c;
        for (int i = 0; i < n_; ++i) {
            // МИНУС по сравнению с delays()
            dt[i] = -R_ * std::cos(theta - phi_[i]) * inv_c;
        }
        return dt;
    }

    std::vector<std::complex<float>>
    UcaGeometry::steeringVector(float freqHz, float thetaDeg, float c) const
    {
        std::vector<std::complex<float>> a(n_);
        const auto tau = delays(thetaDeg, c);
        const float w = TWO_PI_F * freqHz;
        for (int i = 0; i < n_; ++i) {
            const float phase = -w * tau[i];
            a[i] = std::complex<float>(std::cos(phase), std::sin(phase));
        }
        return a;
    }

    int UcaGeometry::nearestElement(float thetaDeg) const {
        const float theta = deg2rad(thetaDeg);
        int best = 0;
        float bestD = 1e9f;
        for (int i = 0; i < n_; ++i) {
            float d = std::fabs(theta - phi_[i]);
            while (d > PI_F) d = std::fabs(d - TWO_PI_F);
            if (d < bestD) { bestD = d; best = i; }
        }
        return best;
    }

} // namespace hydro

#pragma once
#include "types.h"
#include <array>
#include <complex>
#include <vector>

namespace hydro {

// ============================================================
//  Геометрия круговой решётки (UCA) и steering vector.
//
//  Элементы расположены равномерно на окружности радиуса R
//  в горизонтальной плоскости.
//
//  Углы элементов:
//      phi_i = 2*pi*i / N,  i = 0..N-1
//
//  Направление прихода (азимут) отсчитывается от оси +X
//  против часовой стрелки (стандартное соглашение).
//
//  ЗНАК ЗАДЕРЖКИ — ДВА РАЗНЫХ СОГЛАШЕНИЯ:
//
//  1) delays() — для BEAMFORMING (MUSIC/Capon/EV/DAS):
//        tau_i = +R * cos(theta - phi_i) / c
//     Положительное tau — сигнал приходит к элементу ПОЗЖЕ центра.
//     Если источник в направлении theta, то элемент ch_i,
//     ближайший к theta, получает tau с МАКСИМУМОМ (положительным),
//     а противоположный — с минимумом (отрицательным).
//     Steering vector a_i = exp(-j*2*pi*f*tau_i) КОМПЕНСИРУЕТ эту
//     задержку. Используется в алгоритмах DOA.
//
//  2) arrivalOffsets() — для ГЕНЕРАЦИИ СИГНАЛА:
//        dt_i = -R * cos(theta - phi_i) / c
//     Физически: если источник в направлении theta, сигнал приходит
//     к элементу, ближайшему к theta, РАНЬШЕ, чем к центру.
//     Значит, dt_i для него ОТРИЦАТЕЛЬНО. Противоположный элемент
//     получает ПОЛОЖИТЕЛЬНОЕ dt_i (позже).
//     Так s_i(t) = s_source(t - dt_i) даёт корректные фазы.
//
//  ВАЖНО: не перепутать. Для генерации — arrivalOffsets(),
//         для DOA — delays().
// ============================================================

class UcaGeometry {
public:
    UcaGeometry(int numElements = NUM_CHANNELS,
                float radiusMeters = UCA_RADIUS_M);

    int   numElements() const { return n_; }
    float radius()      const { return R_; }

    // Угол i-го элемента (радианы, 0..2*pi)
    float elementAngle(int i) const;

    // Позиция i-го элемента в декартовых координатах (x, y)
    void  elementPosition(int i, float& x, float& y) const;

    // === 1) BEAMFORMING-соглашение ===
    //   tau_i = +R*cos(theta - phi_i) / c
    // Используется в steering vector и DOA.
    std::vector<float> delays(float thetaDeg, float c) const;

    // Steering vector:
    //   a_i = exp(-j * 2*pi*f * tau_i)
    std::vector<std::complex<float>>
        steeringVector(float freqHz, float thetaDeg, float c) const;

    // === 2) ГЕНЕРАЦИЯ-соглашение ===
    //   dt_i = -R*cos(theta - phi_i) / c
    // Используется для задержки сигнала источника на i-м элементе:
    //   s_i(t) = s_source(t - dt_i)
    std::vector<float> arrivalOffsets(float thetaDeg, float c) const;

    // Ближайший элемент к направлению thetaDeg (для визуализации/отладки)
    int nearestElement(float thetaDeg) const;

private:
    int   n_;
    float R_;
    std::vector<float> phi_;   // углы элементов, радианы
};

} // namespace hydro

// ------------------------------------------------------------
//  WAV-писатель (для одиночных тестовых файлов).
// ------------------------------------------------------------
template <typename Channels>
static bool writeWavFloat32(const std::string& path,
                            const Channels& channels,
                            int sampleRate)
{
    if (channels.empty()) return false;
    const uint16_t numChannels = static_cast<uint16_t>(channels.size());
    const size_t numSamples = channels[0].size();
    for (const auto& ch : channels)
    {
        if (ch.size() != numSamples) return false;
    }

    std::ofstream f(path, std::ios::binary);
    if (!f) return false;

    const uint32_t dataBytes = static_cast<uint32_t>(numSamples * numChannels * 4);
    const uint32_t fmtChunkSize = 16;
    const uint16_t audioFormat = 3;
    const uint16_t bitsPerSample = 32;
    const uint32_t byteRate = sampleRate * numChannels * (bitsPerSample / 8);
    const uint16_t blockAlign = numChannels * (bitsPerSample / 8);
    const uint32_t riffSize = 4 + (8 + fmtChunkSize) + (8 + dataBytes);

    auto writeU32 = [&](uint32_t v) { f.write(reinterpret_cast<const char*>(&v), 4); };
    auto writeU16 = [&](uint16_t v) { f.write(reinterpret_cast<const char*>(&v), 2); };

    f.write("RIFF", 4);
    writeU32(riffSize);
    f.write("WAVE", 4);
    f.write("fmt ", 4);
    writeU32(fmtChunkSize);
    writeU16(audioFormat);
    writeU16(numChannels);
    writeU32(sampleRate);
    writeU32(byteRate);
    writeU16(blockAlign);
    writeU16(bitsPerSample);
    f.write("data", 4);
    writeU32(dataBytes);

    std::vector<float> frame(numChannels);
    for (size_t s = 0; s < numSamples; ++s)
    {
        for (uint16_t c = 0; c < numChannels; ++c)
        {
            frame[c] = channels[c][s];
        }
        f.write(reinterpret_cast<const char*>(frame.data()), numChannels * 4);
    }
    return true;
}

// ------------------------------------------------------------
//  WAV-писатель для МОНО (один канал).
// ------------------------------------------------------------
static bool writeMonoWavFloat32(const std::string& path,
                                const std::vector<float>& samples,
                                int sampleRate)
{
    if (samples.empty()) return false;

    std::ofstream f(path, std::ios::binary);
    if (!f) return false;

    const uint16_t numChannels = 1;
    const size_t numSamples = samples.size();
    const uint32_t dataBytes = static_cast<uint32_t>(numSamples * 4);
    const uint32_t fmtChunkSize = 16;
    const uint16_t audioFormat = 3;
    const uint16_t bitsPerSample = 32;
    const uint32_t byteRate = sampleRate * numChannels * (bitsPerSample / 8);
    const uint16_t blockAlign = numChannels * (bitsPerSample / 8);
    const uint32_t riffSize = 4 + (8 + fmtChunkSize) + (8 + dataBytes);

    auto writeU32 = [&](uint32_t v) { f.write(reinterpret_cast<const char*>(&v), 4); };
    auto writeU16 = [&](uint16_t v) { f.write(reinterpret_cast<const char*>(&v), 2); };

    f.write("RIFF", 4);
    writeU32(riffSize);
    f.write("WAVE", 4);
    f.write("fmt ", 4);
    writeU32(fmtChunkSize);
    writeU16(audioFormat);
    writeU16(numChannels);
    writeU32(sampleRate);
    writeU32(byteRate);
    writeU16(blockAlign);
    writeU16(bitsPerSample);
    f.write("data", 4);
    writeU32(dataBytes);

    f.write(reinterpret_cast<const char*>(samples.data()), dataBytes);
    return true;
}

// ------------------------------------------------------------
//  Спектр (DFT) + ASCII-график.
// ------------------------------------------------------------
static std::vector<float> amplitudeSpectrum(const std::vector<float>& x)
{
    const int N = static_cast<int>(x.size());
    const int half = N / 2;
    std::vector<float> spec(half, 0.0f);
    std::vector<float> w(N);
    for (int i = 0; i < N; ++i)
        w[i] = 0.5f * (1.0f - std::cos(TWO_PI_F * i / (N - 1)));

    for (int k = 0; k < half; ++k)
    {
        float re = 0.0f, im = 0.0f;
        const float wk = -TWO_PI_F * k / N;
        for (int n = 0; n < N; ++n)
        {
            const float xn = x[n] * w[n];
            const float ang = wk * n;
            re += xn * std::cos(ang);
            im += xn * std::sin(ang);
        }
        spec[k] = std::sqrt(re * re + im * im) / (N * 0.5f);
    }
    return spec;
}

static void printSpectrumAscii(const std::vector<float>& spectrum,
                               float fs, int fftSize, int topN = 15)
{
    struct Peak
    {
        int bin;
        float freq;
        float amp;
    };
    std::vector<Peak> peaks;
    for (int k = 2; k < static_cast<int>(spectrum.size()) - 1; ++k)
    {
        if (spectrum[k] > spectrum[k - 1] && spectrum[k] > spectrum[k + 1])
            peaks.push_back({k, k * fs / fftSize, spectrum[k]});
    }
    std::sort(peaks.begin(), peaks.end(),
              [](const Peak& a, const Peak& b) { return a.amp > b.amp; });
    if (static_cast<int>(peaks.size()) > topN) peaks.resize(topN);
    std::sort(peaks.begin(), peaks.end(),
              [](const Peak& a, const Peak& b) { return a.freq < b.freq; });

    float maxAmp = 0.0f;
    for (const auto& p : peaks) if (p.amp > maxAmp) maxAmp = p.amp;
    if (maxAmp <= 0.0f) maxAmp = 1.0f;

    std::cout << "  Freq (Hz)  |  Amplitude  |  Bar\n";
    std::cout << "  -----------+-------------+--------------------------------\n";
    for (const auto& p : peaks)
    {
        const int barLen = static_cast<int>(30.0f * p.amp / maxAmp + 0.5f);
        std::string bar(barLen, '#');
        std::cout << "  " << std::setw(9) << std::fixed << std::setprecision(2)
            << p.freq << "  |  "
            << std::setw(9) << std::setprecision(5) << p.amp
            << "  |  " << bar << "\n";
    }
    std::cout << "\n";
}

// ------------------------------------------------------------
//  Вспомогательная: спросить y/n.
// ------------------------------------------------------------
static int askYesNo(const std::string& prompt)
{
    while (true)
    {
        std::cout << prompt << " [y/n/q]: ";
        std::string line;
        if (!std::getline(std::cin, line)) return -1;
        if (line.empty()) continue;
        char c = static_cast<char>(std::tolower(line[0]));
        if (c == 'y') return 1;
        if (c == 'n') return 0;
        if (c == 'q' || c == '0') return -1;
    }
}

        int r = askYesNo("Включить ВИНТ?      ");
        if (r < 0) return;
        useProp = (r == 1);

        r = askYesNo("Включить ДВИГАТЕЛЬ? ");
        if (r < 0) return;
        useEng = (r == 1);

        r = askYesNo("Включить ДЕЛЬФИНА?  ");
        if (r < 0) return;
        useDolphin = (r == 1);

        r = askYesNo("Включить ШУМ ОКЕАНА?");
        if (r < 0) return;
        useOcean = (r == 1);

        if (useProp || useEng || useDolphin || useOcean) break;
        std::cout << "!!! Нужна хотя бы одна компонента. Повторите.\n\n";
    }

    float azimuth = SCENE_AZIMUTH_DEG;
    if (useProp || useEng || useDolphin)
    {
        std::cout << "Азимут (0..359) [Enter = " << SCENE_AZIMUTH_DEG << "]: ";
        std::string line;
        std::getline(std::cin, line);
        if (!line.empty())
        {
            try { azimuth = std::stof(line); }
            catch (...) { azimuth = SCENE_AZIMUTH_DEG; }
        }
    }

    float recvSnr = SCENE_RECEIVER_SNR_DB;
    std::cout << "SNR приёмника (dB) [Enter = " << SCENE_RECEIVER_SNR_DB << "]: ";
    {
        std::string line;
        std::getline(std::cin, line);
        if (!line.empty())
        {
            try { recvSnr = std::stof(line); }
            catch (...) { recvSnr = SCENE_RECEIVER_SNR_DB; }
        }
    }

    const int totalSamples = static_cast<int>(fs) * SCENE_DURATION_SEC;

    SceneParams baseSp;
    baseSp.azimuthDeg = azimuth;
    baseSp.soundSpeed = SOUND_SPEED_WATER;
    baseSp.receiverSnrDb = recvSnr;
    baseSp.noiseSeed = 42;
    baseSp.oceanEnabled = false;

    auto fillOceanParams = [](OceanNoiseParams& op)
    {
        op.level = OCEAN_LEVEL;
        op.rumbleWeight = OCEAN_RUMBLE_WEIGHT;
        op.midWeight = OCEAN_MID_WEIGHT;
        op.hissWeight = OCEAN_HISS_WEIGHT;
        op.rumbleSlope = OCEAN_RUMBLE_SLOPE;
        op.midSlope = OCEAN_MID_SLOPE;
        op.hissSlope = OCEAN_HISS_SLOPE;
        op.spikeRate = OCEAN_SPIKE_RATE;
        op.spikeAmp = OCEAN_SPIKE_AMP;
        op.spikeSpatialFrac = OCEAN_SPIKE_SPATIAL_FRAC;
        op.dropTauMinMs = OCEAN_DROP_TAU_MIN_MS;
        op.dropTauMaxMs = OCEAN_DROP_TAU_MAX_MS;
        op.bubbleTauMinMs = OCEAN_BUBBLE_TAU_MIN_MS;
        op.bubbleTauMaxMs = OCEAN_BUBBLE_TAU_MAX_MS;
        op.bubbleFreqStartMin = OCEAN_BUBBLE_F0_MIN;
        op.bubbleFreqStartMax = OCEAN_BUBBLE_F0_MAX;
        op.bubbleFreqEndMin = OCEAN_BUBBLE_F1_MIN;
        op.bubbleFreqEndMax = OCEAN_BUBBLE_F1_MAX;
        op.clickTauMinMs = OCEAN_CLICK_TAU_MIN_MS;
        op.clickTauMaxMs = OCEAN_CLICK_TAU_MAX_MS;
        op.echoCount = OCEAN_ECHO_COUNT;
        op.echoDelayMinMs = OCEAN_ECHO_DELAY_MIN_MS;
        op.echoDelayMaxMs = OCEAN_ECHO_DELAY_MAX_MS;
        op.echoGainMin = OCEAN_ECHO_GAIN_MIN;
        op.echoGainMax = OCEAN_ECHO_GAIN_MAX;
        op.echoAzSpreadDeg = OCEAN_ECHO_AZ_SPREAD_DEG;
        op.waveRate = OCEAN_WAVE_RATE;
        op.waveAmp = OCEAN_WAVE_AMP;
        op.waveDurationMinS = OCEAN_WAVE_DURATION_MIN_S;
        op.waveDurationMaxS = OCEAN_WAVE_DURATION_MAX_S;
        op.waveSpatialFrac = OCEAN_WAVE_SPATIAL_FRAC;
        op.modulationDepth = OCEAN_MODULATION_DEPTH;
        op.modulationHz = OCEAN_MODULATION_HZ;
        op.seed = OCEAN_SEED;
    };

    auto makeProp = []() -> std::unique_ptr<SourceModel>
    {
        PropellerParams pp;
        pp.numBlades = PROP_NUM_BLADES;
        pp.rpm = PROP_RPM;
        pp.depthM = PROP_DEPTH_M;
        pp.numHarmonics = PROP_NUM_HARMONICS;
        pp.modulationDepth = PROP_MODULATION_DEPTH;
        pp.modulationHz = PROP_MODULATION_HZ;
        pp.cavitation = PROP_CAVITATION;
        pp.snrDb = PROP_SNR_DB;
        pp.seed = PROP_SEED;
        return std::make_unique<PropellerSubmarine>(pp);
    };

    auto makeEng = []() -> std::unique_ptr<SourceModel>
    {
        EngineParams ep;
        ep.rpm = ENG_RPM;
        ep.numHarmonics = ENG_NUM_HARMONICS;
        ep.numResonances = ENG_NUM_RESONANCES;
        ep.mechNoiseLevel = ENG_MECH_NOISE_LEVEL;
        ep.snrDb = ENG_SNR_DB;
        ep.seed = ENG_SEED;
        return std::make_unique<EngineSubmarine>(ep);
    };

    auto makeDolphin = []() -> std::unique_ptr<SourceModel>
    {
        DolphinParams dp;
        dp.numWhistlesPerSec = DOLPHIN_WHISTLES_PER_SEC;
        dp.whistleDurationMin = DOLPHIN_WHISTLE_DUR_MIN;
        dp.whistleDurationMax = DOLPHIN_WHISTLE_DUR_MAX;
        dp.whistleFreqMin = DOLPHIN_WHISTLE_FREQ_MIN;
        dp.whistleFreqMax = DOLPHIN_WHISTLE_FREQ_MAX;
        dp.whistleSweepDepth = DOLPHIN_WHISTLE_SWEEP;
        dp.numHarmonics = DOLPHIN_NUM_HARMONICS;
        dp.clickRate = DOLPHIN_CLICK_RATE;
        dp.clickDurationUs = DOLPHIN_CLICK_DUR_US;
        dp.clickFreqMin = DOLPHIN_CLICK_FREQ_MIN;
        dp.clickFreqMax = DOLPHIN_CLICK_FREQ_MAX;
        dp.clicksPerTrain = DOLPHIN_CLICKS_PER_TRAIN;
        dp.burstRate = DOLPHIN_BURST_RATE;
        dp.burstDurationMin = DOLPHIN_BURST_DUR_MIN;
        dp.burstDurationMax = DOLPHIN_BURST_DUR_MAX;
        dp.burstClickIntervalUs = DOLPHIN_BURST_CLICK_US;
        dp.burstFreqMin = DOLPHIN_BURST_FREQ_MIN;
        dp.burstFreqMax = DOLPHIN_BURST_FREQ_MAX;
        dp.snrDb = DOLPHIN_SNR_DB;
        dp.seed = DOLPHIN_SEED;
        return std::make_unique<Dolphin>(dp);
    };

    if (useProp)
    {
        SceneParams sp = baseSp;
        Scene scene;
        scene.setParams(sp);
        scene.addSource(makeProp());

        std::array<std::vector<float>, NUM_CHANNELS> ch8;
        scene.generate(ch8, totalSamples, fs, uca);

        const std::string name = "scene_propeller.wav";
        if (writeWavFloat32(name, ch8, static_cast<int>(fs)))
            std::cout << "  Saved: " << name << "\n";
        else
            std::cout << "  FAILED: " << name << "\n";
    }

    if (useEng)
    {
        SceneParams sp = baseSp;
        Scene scene;
        scene.setParams(sp);
        scene.addSource(makeEng());

        std::array<std::vector<float>, NUM_CHANNELS> ch8;
        scene.generate(ch8, totalSamples, fs, uca);

        const std::string name = "scene_engine.wav";
        if (writeWavFloat32(name, ch8, static_cast<int>(fs)))
            std::cout << "  Saved: " << name << "\n";
        else
            std::cout << "  FAILED: " << name << "\n";
    }

    if (useDolphin)
    {
        SceneParams sp = baseSp;
        Scene scene;
        scene.setParams(sp);
        scene.addSource(makeDolphin());

        std::array<std::vector<float>, NUM_CHANNELS> ch8;
        scene.generate(ch8, totalSamples, fs, uca);

        const std::string name = "scene_dolphin.wav";
        if (writeWavFloat32(name, ch8, static_cast<int>(fs)))
            std::cout << "  Saved: " << name << "\n";
        else
            std::cout << "  FAILED: " << name << "\n";
    }

    if (useOcean)
    {
        SceneParams sp = baseSp;
        sp.oceanEnabled = true;
        fillOceanParams(sp.oceanParams);

        Scene scene;
        scene.setParams(sp);

        std::array<std::vector<float>, NUM_CHANNELS> ch8;
        scene.generate(ch8, totalSamples, fs, uca);

        const std::string name = "scene_ocean.wav";
        if (writeWavFloat32(name, ch8, static_cast<int>(fs)))
            std::cout << "  Saved: " << name << "\n";
        else
            std::cout << "  FAILED: " << name << "\n";
    }

    {
        SceneParams sp = baseSp;
        if (useOcean)
        {
            sp.oceanEnabled = true;
            fillOceanParams(sp.oceanParams);
        }

        Scene scene;
        scene.setParams(sp);
        if (useProp) scene.addSource(makeProp());
        if (useEng) scene.addSource(makeEng());
        if (useDolphin) scene.addSource(makeDolphin());

        std::array<std::vector<float>, NUM_CHANNELS> ch8;
        scene.generate(ch8, totalSamples, fs, uca);

        const std::string name = "scene_combined.wav";
        if (writeWavFloat32(name, ch8, static_cast<int>(fs)))
            std::cout << "  Saved: " << name << "\n";
        else
            std::cout << "  FAILED: " << name << "\n";

        std::cout << "\n  Spectrum (combined, channel 0, first 8192 samples):\n";
        const int FFT_N = std::min(8192, totalSamples);
        std::vector<float> seg(ch8[0].begin(), ch8[0].begin() + FFT_N);
        auto spec = amplitudeSpectrum(seg);
        printSpectrumAscii(spec, fs, FFT_N, 15);
    }

    std::cout << "\n";
}

// ------------------------------------------------------------
//  ТЕСТОВЫЙ РЕЖИМ: генерация свиста дельфина (моно).
//
//  Новая архитектура: НЕПРЕРЫВНЫЙ длинный ЧМ-сигнал
//  на всю длительность, U-образная траектория частоты,
//  низкие частоты — часть свиста, стрёкот — «шершавый».
// ------------------------------------------------------------
static void generateWhistleTest(float fs)
{
    std::cout << "\n=== ТЕСТ: свист дельфина (моно) ===\n";
    std::cout << "  Модель: DolphinWhistleTest\n";
    std::cout << "  Длительность: 5 сек\n";
    std::cout << "  Частота дискретизации: " << fs << " Гц\n";
    std::cout << "  Формат: моно, 32-bit float\n\n";

    DolphinWhistleTestParams p;
    p.fs = fs;
    p.durationSec = 5.0f;
    p.seed = 77777;
    // Всё остальное — из .h (дефолты).

    DolphinWhistleTest gen(p);

    const int totalSamples = static_cast<int>(fs * p.durationSec);
    std::vector<float> buffer(totalSamples, 0.0f);

    gen.generate(buffer.data(), totalSamples);

    const std::string name = "whistle_test.wav";
    if (writeMonoWavFloat32(name, buffer, static_cast<int>(fs)))
        std::cout << "  Saved: " << name << "\n";
    else
        std::cout << "  FAILED: " << name << "\n";

    std::cout << "  Прогоните " << name << " через sound_digit и пришлите отчёт.\n\n";
}

// ------------------------------------------------------------
//  Режим "2" — пакетный датасет.
// ------------------------------------------------------------
static void generateDataset(const UcaGeometry& uca, float fs)
{
    std::cout << "\n=== Генерация датасета ===\n";

    DatasetSelection sel;
    while (true)
    {
        int r = askYesNo("Генерировать PropellerSub?");
        if (r < 0) return;
        sel.propeller = (r == 1);

        r = askYesNo("Генерировать EngineSub?   ");
        if (r < 0) return;
        sel.engine = (r == 1);

        r = askYesNo("Генерировать Dolphin?     ");
        if (r < 0) return;
        sel.dolphin = (r == 1);

        r = askYesNo("Генерировать OceanNoise?  ");
        if (r < 0) return;
        sel.ocean = (r == 1);

        if (sel.propeller || sel.engine || sel.dolphin || sel.ocean) break;
        std::cout << "!!! Хотя бы один класс должен быть выбран. Повторите.\n\n";
    }

    DatasetParams dp;
    dp.uniformAzimuth = true;
    dp.azimuthDeg = SCENE_AZIMUTH_DEG;

    if (sel.propeller || sel.engine || sel.dolphin)
    {
        int r = askYesNo("Равномерный азимут 0..359?  ");
        if (r < 0) return;
        dp.uniformAzimuth = (r == 1);

        if (!dp.uniformAzimuth)
        {
            std::cout << "Конкретный азимут (0..359) [Enter = " << SCENE_AZIMUTH_DEG << "]: ";
            std::string line;
            std::getline(std::cin, line);
            if (!line.empty())
            {
                try { dp.azimuthDeg = std::stof(line); }
                catch (...) { dp.azimuthDeg = SCENE_AZIMUTH_DEG; }
            }
        }
    }

    dp.receiverSnrDb = SCENE_RECEIVER_SNR_DB;
    std::cout << "SNR приёмника (dB) [Enter = " << SCENE_RECEIVER_SNR_DB << "]: ";
    {
        std::string line;
        std::getline(std::cin, line);
        if (!line.empty())
        {
            try { dp.receiverSnrDb = std::stof(line); }
            catch (...) { dp.receiverSnrDb = SCENE_RECEIVER_SNR_DB; }
        }
    }

    int n = 0;
    std::cout << "Сколько файлов на класс? [Enter = 10]: ";
    std::string line;
    std::getline(std::cin, line);
    if (!line.empty())
    {
        try { n = std::stoi(line); }
        catch (...) { n = 0; }
    }
    if (n <= 0) n = 10;

    dp.filesPerClass = n;
    dp.durationSec = SCENE_DURATION_SEC;
    dp.sampleRate = fs;
    dp.outputDir = "dataset";

    DatasetGenerator gen(uca, dp, sel);
    gen.generate();
}

// ------------------------------------------------------------
//  main
// ------------------------------------------------------------
int main()
{
#ifdef _WIN32
    SetConsoleOutputCP(CP_UTF8);
    SetConsoleCP(CP_UTF8);
#endif

    std::cout << "=== hydroacoustic ===\n";

    UcaGeometry uca(NUM_CHANNELS, UCA_RADIUS_M);
    const float fs = static_cast<float>(SAMPLE_RATE);

    while (true)
    {
        std::cout << "\n";
        std::cout << "  1 - Сгенерировать одиночный WAV (интерактивный выбор)\n";
        std::cout << "  2 - Сгенерировать пакетный датасет\n";
        std::cout << "  3 - ТЕСТ: свист дельфина (моно, отдельная модель)\n";
        std::cout << "  0 - Выход\n";
        std::cout << "Выбор: ";

        std::string choice;
        if (!std::getline(std::cin, choice)) break;
        if (choice == "0") break;

        if (choice == "1")
        {
            generateSingleFiles(uca, fs);
        }
        else if (choice == "2")
        {
            generateDataset(uca, fs);
        }
        else if (choice == "3")
        {
            generateWhistleTest(fs);
        }
        else
        {
            std::cout << "Неизвестный выбор.\n";
        }
    }

    std::cout << "Bye.\n";
    return 0;
}

    // ============================================================
    //  Рендер одной фразы.
    //
    //  Основная F0 плавно едет по форме (нисходящая / восходящая / U).
    //  В хвосте (u > tailStartFrac) F0 уходит в низ (до fEnd × tailEndFrac).
    //  На каждом отсчёте — 3 гармоники (2×, 3×, 4×) с базовыми
    //  уровнями harm1Db/harm2Db/harm3Db.
    // ============================================================
    void DolphinWhistleTest::renderPhrase(std::vector<float>& buffer,
                                          int numSamples,
                                          Phrase& ph)
    {
        if (ph.lengthSamples <= 0) return;

        const float fs = p_.fs;
        const float invFs = 1.0f / fs;
        const float durSec = static_cast<float>(ph.lengthSamples) * invFs;

        const float h1 = std::pow(10.0f, p_.harm1Db / 20.0f);
        const float h2 = std::pow(10.0f, p_.harm2Db / 20.0f);
        const float h3 = std::pow(10.0f, p_.harm3Db / 20.0f);

        float phaseMain = ph.phase;
        float phaseH[3] = {
            ph.phase * 0.7f, ph.phase * 0.3f, ph.phase * 0.5f
        };

        ph.fInstTrace.assign(ph.lengthSamples, 0.0f);

        for (int n = 0; n < ph.lengthSamples; ++n)
        {
            const int idx = ph.sampleStart + n;
            if (idx < 0 || idx >= numSamples) continue;

            const float tLocal = static_cast<float>(n) * invFs;
            const float u = tLocal / durSec;

            // --- Основная F0 по форме ---
            float fBase;
            const float s = 0.5f * (1.0f - std::cos(PI_F * u));

            if (ph.shape == PhraseShape::Descending)
                fBase = ph.fStart + (ph.fEnd - ph.fStart) * s;
            else if (ph.shape == PhraseShape::Ascending)
                fBase = ph.fEnd + (ph.fStart - ph.fEnd) * s;
            else // UShape
            {
                const float mid = 0.5f * (ph.fStart + ph.fEnd);
                if (u < 0.5f)
                {
                    const float v = u * 2.0f;
                    const float sv = 0.5f * (1.0f - std::cos(PI_F * v));
                    fBase = ph.fStart + (mid - ph.fStart) * sv;
                }
                else
                {
                    const float v = (u - 0.5f) * 2.0f;
                    const float sv = 0.5f * (1.0f - std::cos(PI_F * v));
                    fBase = mid + (ph.fStart - mid) * sv;
                }
            }

            // --- Хвост: уход в низ ---
            float fInst = fBase;
            const bool inTail = (u > p_.tailStartFrac);
            if (inTail)
            {
                const float v = (u - p_.tailStartFrac)
                    / (1.0f - p_.tailStartFrac);
                const float sv = 0.5f * (1.0f - std::cos(PI_F * v));
                const float fTail = ph.fEnd * p_.tailEndFrac;
                fInst = fBase + (fTail - fBase) * sv;
            }

            // --- Вибрато ---
            const float vibMain = p_.vibDepthHz
                * std::sin(TWO_PI_F * p_.vibRateHz * tLocal);
            fInst += vibMain;

            ph.fInstTrace[n] = fInst;

            // --- Огибающая ---
            float env = 1.0f;
            if (u < p_.riseFrac)
                env = 0.5f * (1.0f - std::cos(PI_F * u / p_.riseFrac));
            else if (u > 1.0f - p_.fallFrac)
            {
                const float v = (1.0f - u) / p_.fallFrac;
                env = 0.5f * (1.0f - std::cos(PI_F * v));
            }

            // --- AM ---
            const float am = 1.0f
                - p_.amDepth * 0.5f
                + p_.amDepth * 0.5f
                * std::sin(TWO_PI_F * p_.amRateHz * tLocal);

            // --- Основной тон ---
            phaseMain += TWO_PI_F * fInst * invFs;
            if (phaseMain > TWO_PI_F) phaseMain -= TWO_PI_F;
            float sig = std::sin(phaseMain);

            // Гармоники 2×, 3×, 4× — простые, тихие.
            // Стрёкот делается отдельно, в renderCrackleForPhrase.
            const float gains[3] = {h1, h2, h3};
            for (int k = 0; k < 3; ++k)
            {
                const float fH = fInst * float(k + 2);
                if (fH >= fs * 0.45f) break;

                phaseH[k] += TWO_PI_F * fH * invFs;
                if (phaseH[k] > TWO_PI_F) phaseH[k] -= TWO_PI_F;

                sig += gains[k] * std::sin(phaseH[k]);
            }

            buffer[idx] += sig * env * am;
        }
    }

    // ============================================================
    //  Эхо одной фразы.
    //  Повторяет F0-траекторию, но тише, чуть ниже и растянуто.
    // ============================================================
    void DolphinWhistleTest::renderEchoForPhrase(std::vector<float>& buffer,
                                                 int numSamples,
                                                 const Phrase& ph)
    {
        if (ph.fInstTrace.empty()) return;

        const float fs = p_.fs;
        const float invFs = 1.0f / fs;

        const int nEcho = std::max(1, p_.echoCount);
        const float gains[3] = {
            std::pow(10.0f, p_.echoGain1Db / 20.0f),
            std::pow(10.0f, p_.echoGain2Db / 20.0f),
            std::pow(10.0f, p_.echoGain3Db / 20.0f)
        };

        for (int e = 0; e < nEcho && e < 1; ++e)
        {
            const float delay = p_.echoDelayMinMs * 0.001f;
            const int delaySamples = static_cast<int>(delay * fs);
            const float gain = gains[e];
            const float pitch = p_.echoPitchFactor;
            const float stretch = p_.echoStretch;

            const int echoLen = static_cast<int>(ph.lengthSamples * stretch);
            const float invEchoDur = 1.0f
                / (static_cast<float>(echoLen) * invFs);

            float phaseMain = ph.phase * 0.5f;
            float phaseH[2] = {ph.phase * 0.3f, ph.phase * 0.8f};

            for (int n = 0; n < echoLen; ++n)
            {
                const int idx = ph.sampleStart + delaySamples + n;
                if (idx < 0 || idx >= numSamples) continue;

                const float tLocal = static_cast<float>(n) * invFs;
                const float u = tLocal * invEchoDur;

                const float srcU = u / stretch;
                if (srcU < 0.0f || srcU >= 1.0f) continue;
                const int srcIdx = static_cast<int>(srcU
                    * static_cast<float>(ph.lengthSamples));
                if (srcIdx < 0
                    || srcIdx >= int(ph.fInstTrace.size()))
                    continue;

                const float fInst = ph.fInstTrace[srcIdx] * pitch;
                const float env = std::exp(-u * 1.5f);

                phaseMain += TWO_PI_F * fInst * invFs;
                if (phaseMain > TWO_PI_F) phaseMain -= TWO_PI_F;
                float sig = std::sin(phaseMain);

                for (int k = 0; k < 2 && k < p_.numHarmonics; ++k)
                {
                    const float fH = fInst * float(k + 2);
                    if (fH >= fs * 0.45f) break;
                    phaseH[k] += TWO_PI_F * fH * invFs;
                    if (phaseH[k] > TWO_PI_F) phaseH[k] -= TWO_PI_F;
                    const float a = 0.4f / float(k + 1);
                    sig += a * std::sin(phaseH[k]);
                }

                buffer[idx] += sig * env * gain;
            }
        }
    }


    // ============================================================
    //  Стрёкот — ШУМОВАЯ АРХИТЕКТУРА (правильный BPF).
    //
    //  Шумовой источник — сумма синусов на случайных частотах
    //  в полосе 4–8 кГц. Это даёт «звонкость» в П3.
    //
    //  Амплитуда пульсирует серией коротких гауссовых импульсов
    //  с частотой 30–60 Гц. Каждый импульс — 1–3 мс.
    // ============================================================
    void DolphinWhistleTest::renderCrackleForPhrase(
        std::vector<float>& buffer, int numSamples, const Phrase& ph)
    {
        if (!ph.hasCrackle) return;
        if (p_.crackleLevel <= 0.0f) return;

        const float fs = p_.fs;
        const float invFs = 1.0f / fs;

        // Начало стрёкота — с crackleStartU фразы, до конца.
        const int crackleStart = ph.sampleStart
            + static_cast<int>(ph.lengthSamples * p_.crackleStartU);
        const int crackleEnd = ph.sampleStart + ph.lengthSamples;

        const int totalLen = crackleEnd - crackleStart;
        if (totalLen <= 0) return;

        const float totalDurSec = static_cast<float>(totalLen) * invFs;

        // --- Параметры серии (случайные) ---
        std::uniform_real_distribution<float> pulseDist(
            p_.cracklePulseHzMin, p_.cracklePulseHzMax);
        std::uniform_real_distribution<float> phaseDist(0.0f, TWO_PI_F);

        const float pulseHz = pulseDist(rng_);

        // --- Шумовой источник: сумма синусов в полосе 4–8 кГц ---
        // 16 синусов на случайных частотах в полосе.
        constexpr int N_SINES = 16;
        std::vector<float> sineFreqs(N_SINES);
        std::vector<float> sinePhases(N_SINES);
        std::uniform_real_distribution<float> freqDist(
            p_.crackleBpfLowHz, p_.crackleBpfHighHz);
        for (int k = 0; k < N_SINES; ++k)
        {
            sineFreqs[k] = freqDist(rng_);
            sinePhases[k] = phaseDist(rng_);
        }

        // --- Серия гауссовых импульсов ---
        // Моменты импульсов — с частотой pulseHz.
        std::vector<float> pulseCenters;
        {
            const float period = 1.0f / pulseHz;
            float t = 0.0f;
            while (t < totalDurSec)
            {
                pulseCenters.push_back(t);
                t += period;
            }
        }

        // Длительность импульса (гауссова) — 1–3 мс.
        // Берём среднее — 2 мс, tau = 1 мс.
        const float tau = 0.001f;
        const float invTau2 = 1.0f / (tau * tau);

        // --- Рендер ---
        for (int n = 0; n < totalLen; ++n)
        {
            const int idx = crackleStart + n;
            if (idx < 0 || idx >= numSamples) continue;

            const float tLocal = static_cast<float>(n) * invFs;
            const float u = tLocal / totalDurSec;

            // --- Огибающая серии ---
            float env = 1.0f;
            if (u < p_.crackleRiseFrac)
                env = 0.5f * (1.0f - std::cos(PI_F * u
                    / p_.crackleRiseFrac));
            else if (u > 1.0f - p_.crackleFallFrac)
            {
                const float v = (1.0f - u) / p_.crackleFallFrac;
                env = 0.5f * (1.0f - std::cos(PI_F * v));
            }

            // --- Пульсирующая огибающая ---
            float pulseEnv = 0.0f;
            for (float pc : pulseCenters)
            {
                const float dt = tLocal - pc;
                if (dt < -0.005f || dt > 0.005f) continue;
                const float g = std::exp(-(dt * dt) * invTau2);
                if (g > pulseEnv) pulseEnv = g;
            }
            if (pulseEnv < 1e-4f) continue;

            // --- Шумовой источник: сумма синусов ---
            float carrier = 0.0f;
            for (int k = 0; k < N_SINES; ++k)
            {
                carrier += std::sin(TWO_PI_F * sineFreqs[k]
                    * tLocal + sinePhases[k]);
            }
            // Нормировка: 1/sqrt(N), чтобы RMS не рос.
            carrier /= std::sqrt(static_cast<float>(N_SINES));

            // --- Итог ---
            buffer[idx] += carrier * pulseEnv * env * p_.crackleLevel;
        }
    }

    // ============================================================
    //  Шум моря — очень тихий, HP-фильтр (без НЧ-гула).
    // ============================================================
    void DolphinWhistleTest::renderSeaNoise(std::vector<float>& buffer,
                                            int numSamples)
    {
        if (p_.seaNoiseLevel <= 0.0f) return;

        const float fs = p_.fs;
        const float invFs = 1.0f / fs;
        const float fc = p_.seaNoiseHP;
        const float a = std::exp(-TWO_PI_F * fc * invFs);
        float lp = 0.0f;

        for (int n = 0; n < numSamples; ++n)
        {
            const float t = static_cast<float>(n) * invFs;

            const float white = gauss_(rng_);
            lp = a * lp + (1.0f - a) * white;
            const float hp = white - lp;

            const float am = 1.0f
                - p_.seaNoiseAmDepth * 0.5f
                + p_.seaNoiseAmDepth * 0.5f
                * std::sin(TWO_PI_F * p_.seaNoiseAmHz * t);

            buffer[n] += hp * am * p_.seaNoiseLevel;
        }
    }

    // ============================================================
    //  Главный метод.
    // ============================================================
    void DolphinWhistleTest::generate(float* out, int numSamples)
    {
        if (numSamples <= 0) return;

        for (int n = 0; n < numSamples; ++n) out[n] = 0.0f;

        std::vector<float> buffer(numSamples, 0.0f);
        rng_.seed(p_.seed);

        const float fs = p_.fs;
        const float totalDurSec = static_cast<float>(numSamples) / fs;

        std::uniform_real_distribution<float> phraseDurDist(
            p_.phraseDurMin, p_.phraseDurMax);
        std::uniform_real_distribution<float> pauseDist(
            p_.pauseMinMs * 0.001f, p_.pauseMaxMs * 0.001f);
        std::uniform_real_distribution<float> fStartDist(
            p_.fStartMin, p_.fStartMax);
        std::uniform_real_distribution<float> fEndDist(
            p_.fEndMin, p_.fEndMax);
        std::uniform_real_distribution<float> phaseDist(0.0f, TWO_PI_F);
        std::uniform_real_distribution<float> startSilenceDist(
            p_.startSilenceMin, p_.startSilenceMax);
        std::uniform_real_distribution<float> endSilenceDist(
            p_.endSilenceMin, p_.endSilenceMax);
        std::uniform_real_distribution<float> uniform01(0.0f, 1.0f);

        const float tStart = startSilenceDist(rng_);
        const float tEnd = totalDurSec - endSilenceDist(rng_);

        std::vector<Phrase> phrases;

        // --- Основные фразы ---
        float t = tStart;
        while (t < tEnd && int(phrases.size()) < p_.phraseCount)
        {
            const float phraseDur = phraseDurDist(rng_);
            if (t + phraseDur > tEnd) break;

            Phrase ph;
            ph.sampleStart = static_cast<int>(t * fs);
            ph.lengthSamples = static_cast<int>(phraseDur * fs);
            ph.fStart = fStartDist(rng_);
            ph.fEnd = fEndDist(rng_);
            ph.phase = phaseDist(rng_);
            ph.hasCrackle = (uniform01(rng_) < p_.crackleProb);
            ph.shape = PhraseShape::Descending;

            phrases.push_back(ph);
            t += phraseDur + pauseDist(rng_);
        }

        // --- Альтернативные фразы (накладываются) ---
        for (int k = 0; k < p_.altPhraseCount; ++k)
        {
            const float phraseDur = phraseDurDist(rng_);
            const float tRange = tEnd - tStart - phraseDur;
            if (tRange <= 0.0f) break;
            const float tPos = tStart + uniform01(rng_) * tRange;

            Phrase ph;
            ph.sampleStart = static_cast<int>(tPos * fs);
            ph.lengthSamples = static_cast<int>(phraseDur * fs);
            ph.fStart = fStartDist(rng_);
            ph.fEnd = fEndDist(rng_);
            ph.phase = phaseDist(rng_);
            ph.hasCrackle = (uniform01(rng_) < p_.crackleProb);
            ph.shape = (k % 2 == 0)
                           ? PhraseShape::UShape
                           : PhraseShape::Ascending;

            phrases.push_back(ph);
        }

        renderSeaNoise(buffer, numSamples);

        for (auto& ph : phrases)
        {
            renderPhrase(buffer, numSamples, ph);
            renderEchoForPhrase(buffer, numSamples, ph);
            renderCrackleForPhrase(buffer, numSamples, ph);
        }

        // --- Нормализация по пику к 0.95 ---
        float maxAbs = 0.0f;
        for (int n = 0; n < numSamples; ++n)
        {
            const float a = std::fabs(buffer[n]);
            if (a > maxAbs) maxAbs = a;
        }
        const float scale = (maxAbs > 1e-9f) ? (0.95f / maxAbs) : 1.0f;

        for (int n = 0; n < numSamples; ++n)
            out[n] = buffer[n] * scale;
    }

        float fallFrac = 0.25f;

        // --- Тембр ---

        float harm1Db = -12.0f;
        float harm2Db = -18.0f;
        float harm3Db = -24.0f;
        float harm4Db = -30.0f;

        // --- Вибрато ---

        float vibRateHz = 6.0f;

        // --- AM ---

        float amRateHz = 3.0f;

        // --- Паузы ---
        float pauseMinMs = 200.0f;
        float pauseMaxMs = 500.0f;

        // --- Тишина ---
        float startSilenceMin = 0.3f;
        float startSilenceMax = 0.6f;
        float endSilenceMin = 0.3f;
        float endSilenceMax = 0.6f;

        // --- Эхо ---
        int echoCount = 1;
        float echoDelayMinMs = 80.0f;
        float echoDelayMaxMs = 80.0f;
        float echoGain1Db = -15.0f;
        float echoGain2Db = -22.0f;
        float echoGain3Db = -28.0f;

        float echoStretch = 1.0f;

        //
        //  Не тон. Шум. BPF 4–8 кГц + пульсирующая огибающая
        //  с частотой 30–60 Гц. Это — «треск».
        // ---
        // ============================================================

        // ------------------------------------------------------------
        //  State Variable Filter (Chamberlin). Стабильнее biquad
        //  на краях диапазона, даёт сразу LP/BP/HP выходы.
        //
        //  f  = 2*sin(pi * fc / fs) — нормированная частота
        //  q  = 1/Q                — затухание
        //
        //  Коэффициенты пересчитываются при set().
        // ------------------------------------------------------------
        struct SVF
        {
            float f = 0.1f;
            float q = 1.0f;
            float lp = 0.0f;
            float bp = 0.0f;

            void set(float fs, float fc, float Q)
            {
                if (fc < 5.0f) fc = 5.0f;
                if (fc > fs * 0.45f) fc = fs * 0.45f;
                if (Q < 0.5f) Q = 0.5f;
                if (Q > 4.0f) Q = 4.0f;

                f = 2.0f * std::sin(PI_F * fc / fs);
                if (f > 1.0f) f = 1.0f;
                q = 1.0f / Q;
            }

            float processBP(float x)
            {
                const float hp = x - lp - q * bp;
                bp += f * hp;
                lp += f * bp;

                if (!std::isfinite(bp) || std::fabs(bp) > 1e6f)
                {
                    bp = 0.0f;
                    lp = 0.0f;
                }
                if (!std::isfinite(lp) || std::fabs(lp) > 1e6f)
                {
                    lp = 0.0f;
                }

                return bp;
            }

            void reset()
            {
                lp = 0.0f;
                bp = 0.0f;
            }
        };

        // ------------------------------------------------------------
        //  Огибающая: attack + exponential decay.
        //
        //  t       — время с начала события (сек)
        //  tAttack — длительность нарастания
        //  tDecay  — характерное время спада (1/e)
        //
        //  Возвращает 0 за пределами [0, tAttack + 5*tDecay].
        // ------------------------------------------------------------
        inline float envelope(float t, float tAttack, float tDecay)
        {
            if (t < 0.0f) return 0.0f;
            const float tEnd = tAttack + 5.0f * tDecay;
            if (t > tEnd) return 0.0f;

            if (t < tAttack)
            {
                // Плавное нарастание — половина косинуса (без излома).
                return 0.5f * (1.0f - std::cos(PI_F * t / tAttack));
            }
            const float u = (t - tAttack) / tDecay;
            return std::exp(-u);
        }

        // ------------------------------------------------------------
        //  Один слой: полосовой шум × огибающая, СРАЗУ по 8 каналам.
        //
        //  Для каждого канала — свой белый шум, свой SVF, своя задержка.
        //  Это даёт настоящий пространственный эффект: каналы
        //  некоррелированы, но огибающая общая (одна волна).
        // ------------------------------------------------------------
        // ------------------------------------------------------------
        //  Один слой: полосовой шум × огибающая, СРАЗУ по 8 каналам.
        //
        //  Ключевые механизмы стерео:
        //   1. Общий band — «скелет» волны, один на все каналы.
        //   2. Направленность g(θ, φ_ch) = 0.5 + 0.5*cos(θ - φ_ch):
        //      ближний к азимуту канал громче, дальний — тише.
        //      При θ(t) амплитуда «перетекает» между каналами.
        //   3. Субдискретная задержка: линейная интерполяция band.
        //   4. Малый независимый шум приёмника на каждый канал.
        // ------------------------------------------------------------
        void renderLayer(std::array<std::vector<float>, NUM_CHANNELS>& out,
                         int numSamples,
                         float fs,
                         const OceanNoise::Wave& w,
                         const UcaGeometry& uca,
                         float soundSpeed,
                         float fLow, float fHigh,
                         float tAttack, float tDecay,
                         float tDelay,
                         float amp,
                         std::mt19937& rng)
        {
            if (w.lengthSamples <= 0) return;

            const float fc = std::sqrt(fLow * fHigh);
            const float bw = (fHigh - fLow);
            const float Q = (bw > 1.0f) ? (fc / bw) : 1.0f;
            const float invFs = 1.0f / fs;
            const int delaySamples = static_cast<int>(tDelay * fs);

            std::normal_distribution<float> gauss(0.0f, 1.0f);

            // SVF для общего band.
            SVF svfMain;
            svfMain.set(fs, fc, Q);

            // SVF для шума приёмника — по одному на канал.
            SVF svfNoise[NUM_CHANNELS];
            for (int ch = 0; ch < NUM_CHANNELS; ++ch)
                svfNoise[ch].set(fs, fc, Q);

            const bool spatial = (w.azimuthDeg >= 0.0f);
            const float az0 = w.azimuthDeg;
            const float invC = 1.0f / soundSpeed;

            // φ каналов.
            float phi[NUM_CHANNELS];
            for (int ch = 0; ch < NUM_CHANNELS; ++ch)
                phi[ch] = TWO_PI_F * ch / NUM_CHANNELS;

            // Кольцевой буфер последних значений band для интерполяции.
            // Размер — с запасом: макс. задержка = 2R/c * fs < 10 отсч.
            constexpr int HIST = 32;
            float bandHist[HIST] = {0.0f};
            int histPos = 0;

            constexpr float RECEIVER_NOISE_MIX = 0.20f;

            for (int k = 0; k < w.lengthSamples; ++k)
            {
                const int idx = w.sampleIndex + k;
                if (idx < 0 || idx >= numSamples) continue;

                const float tEnv = static_cast<float>(k - delaySamples) * invFs;
                const float env = envelope(tEnv, tAttack, tDecay);

                // --- Общий band (генерируем всегда, даже если env=0) ---
                const float white = gauss(rng);
                const float band = svfMain.processBP(white);

                // Пишем в кольцевой буфер.
                bandHist[histPos] = band;
                histPos = (histPos + 1) % HIST;

                // Прогрев шума приёмника (нужен всегда, чтобы не было щелчка).
                float noiseBand[NUM_CHANNELS];
                for (int ch = 0; ch < NUM_CHANNELS; ++ch)
                {
                    const float wn = gauss(rng);
                    noiseBand[ch] = svfNoise[ch].processBP(wn);
                }

                if (env <= 0.0f) continue;

                const float tLocal = static_cast<float>(k) * invFs;

                if (spatial)
                {
                    // θ(t) — плавающий азимут.
                    const float azNow = az0 + w.azSweepDeg *
                        std::sin(TWO_PI_F * w.azSweepHz * tLocal + w.azSweepPhase);
                    const float theta = deg2rad(azNow);

                    for (int ch = 0; ch < NUM_CHANNELS; ++ch)
                    {
                        // --- Направленность: ближний к θ канал громче ---
                        // g = 0.5 + 0.5*cos(θ - φ_ch) ∈ [0, 1]
                        const float gDir = 0.5f + 0.5f *
                            std::cos(theta - phi[ch]);

                        // --- Субдискретная задержка ---
                        // Сколько отсчётов назад «смотрит» этот канал.
                        const float dS = (-uca.radius() *
                            std::cos(theta - phi[ch]) * invC) * fs;

                        // Задержка — обычно маленькая (0..10 отсч.),
                        // может быть отрицательной (опережение).
                        // Разложим на целую и дробную часть.
                        int dInt = static_cast<int>(std::floor(dS));
                        float dFrac = dS - static_cast<float>(dInt);

                        // Ограничиваем, чтобы не выйти за HIST.
                        if (dInt > HIST - 2) dInt = HIST - 2;
                        if (dInt < -1) dInt = -1;

                        // Смотрим в кольцевой буфер назад на dInt отсчётов.
                        // histPos указывает на «следующую свободную» ячейку,
                        // значит последний записанный — (histPos-1+HIST)%HIST.
                        auto histAt = [&](int back) -> float
                        {
                            // back >= 0 — назад на back отсчётов от последнего.
                            int p = (histPos - 1 - back + 2 * HIST) % HIST;
                            return bandHist[p];
                        };

                        // band с задержкой dS = (1 - dFrac) * band[dInt] +
                        //                        dFrac * band[dInt + 1]
                        // где band[dInt] — это signal на dInt отсчётов назад.
                        const float b0 = histAt(dInt < 0 ? 0 : dInt);
                        const float b1 = histAt(dInt < 0 ? 0 : dInt + 1);
                        const float delayed = (1.0f - dFrac) * b0 +
                            dFrac * b1;

                        // Пишем: направленность × задержанный сигнал
                        //       + малый шум приёмника.
                        const float main = delayed * gDir * (1.0f - RECEIVER_NOISE_MIX);
                        const float noise = noiseBand[ch] * RECEIVER_NOISE_MIX;
                        out[ch][idx] += amp * env * (main + noise);
                    }
                }
                else
                {
                    // Без азимута: общий band + шум приёмника в каждый канал.
                    for (int ch = 0; ch < NUM_CHANNELS; ++ch)
                    {
                        const float main = band * (1.0f - RECEIVER_NOISE_MIX);
                        const float noise = noiseBand[ch] * RECEIVER_NOISE_MIX;
                        out[ch][idx] += amp * env * (main + noise);
                    }
                }
            }
        }

    // ============================================================
    //  Фоновый шум: три слоя (rumble/mid/hiss) — как было.
    // ============================================================
    void OceanNoise::generateBackgroundChannel(std::vector<float>& outChannel,
                                               int numSamples,
                                               float /*fs*/,
                                               uint32_t channelSeed)
    {
        outChannel.assign(numSamples, 0.0f);
        if (p_.rumbleWeight <= 0.0f && p_.midWeight <= 0.0f && p_.hissWeight <= 0.0f)
            return;

        std::mt19937 rng(channelSeed);
        std::normal_distribution<float> gauss(0.0f, 1.0f);
        float rS = 0.0f, mS = 0.0f, hS = 0.0f;
        const float wR = p_.rumbleWeight, wM = p_.midWeight, wH = p_.hissWeight;
        const float norm = 1.0f / std::sqrt(wR * wR + wM * wM + wH * wH + 1e-6f);

        for (int n = 0; n < numSamples; ++n)
        {
            const float white = gauss(rng);

            // Однополюсные ФНЧ (как было).
            rS = p_.rumbleSlope * rS + (1.0f - p_.rumbleSlope) * white;
            mS = p_.midSlope * mS + (1.0f - p_.midSlope) * white;
            hS = p_.hissSlope * hS + (1.0f - p_.hissSlope) * white;

            const float gR = 1.0f / std::sqrt(1.0f - p_.rumbleSlope * p_.rumbleSlope + 1e-6f);
            const float gM = 1.0f / std::sqrt(1.0f - p_.midSlope * p_.midSlope + 1e-6f);
            const float gH = 1.0f / std::sqrt(1.0f - p_.hissSlope * p_.hissSlope + 1e-6f);

            const float r = rS * gR * wR;
            const float m = mS * gM * wM;
            const float h = hS * gH * wH;
            outChannel[n] = (r + m + h) * norm * p_.level;
        }
    }

    // ============================================================
    //  Всплески — БЕЗ ИЗМЕНЕНИЙ.
    // ============================================================
    void OceanNoise::renderSpike(std::vector<float>& buffer,
                                 int numSamples,
                                 float fs,
                                 int startSample,
                                 const OceanNoise::Spike& s,
                                 std::mt19937& rng)
    {
        std::normal_distribution<float> gauss(0.0f, 1.0f);
        const float tau = std::max(1e-4f, s.tauSec);
        const int len = std::min(numSamples - startSample,
                                 static_cast<int>(5.0f * tau * fs) + 1);

        for (int k = 0; k < len; ++k)
        {
            const float t = static_cast<float>(k) / fs;
            const float env = std::exp(-(t / tau) * (t / tau));
            float sig = 0.0f;
            if (s.type == 1)
            {
                const float T = 3.0f * tau;
                const float u = std::min(1.0f, t / T);
                const float f = s.f0 + (s.f1 - s.f0) * u;
                sig = std::sin(TWO_PI_F * f * t + s.phase);
            }
            else
            {
                sig = gauss(rng);
            }
            buffer[startSample + k] += s.amplitude * env * sig;
        }
    }

    // ============================================================
    //  Волна: 4 слоя полосового шума с индивидуальными огибающими.
    //
    //  Это — ключевая замена. Было: 4 синуса × Hann.
    //  Стало: 4 полосовых шумовых слоя, у каждого attack/decay
    //         и сдвиг по времени (пена и шелест запаздывают).
    // ============================================================
    void OceanNoise::renderWave(std::array<std::vector<float>, NUM_CHANNELS>& out,
                                int numSamples,
                                float fs,
                                const OceanNoise::Wave& w,
                                const OceanNoiseParams& p,
                                const UcaGeometry& uca,
                                float soundSpeed,
                                std::mt19937& rng)
    {
        if (w.lengthSamples <= 0) return;

        const float A = w.amplitude;

        renderLayer(out, numSamples, fs, w, uca, soundSpeed,
                    p.rumbleLow, p.rumbleHigh,
                    p.rumbleAttack, p.rumbleDecay,
                    0.0f, A * p.rumbleMix, rng);

        renderLayer(out, numSamples, fs, w, uca, soundSpeed,
                    p.bodyLow, p.bodyHigh,
                    p.bodyAttack, p.bodyDecay,
                    0.0f, A * p.bodyMix, rng);

        renderLayer(out, numSamples, fs, w, uca, soundSpeed,
                    p.foamLow, p.foamHigh,
                    p.foamAttack, p.foamDecay,
                    p.foamDelay, A * p.foamMix, rng);

        renderLayer(out, numSamples, fs, w, uca, soundSpeed,
                    p.hissLow, p.hissHigh,
                    p.hissAttack, p.hissDecay,
                    p.hissDelay, A * p.hissMix, rng);
    }

    // ============================================================
    //  Генерация спайков — БЕЗ ИЗМЕНЕНИЙ.
    // ============================================================
    std::vector<OceanNoise::Spike>
    OceanNoise::generateSpikes(int numSamples, float fs, uint32_t seed)
    {
        std::vector<Spike> spikes;
        if (p_.spikeRate <= 0.0f || p_.spikeAmp <= 0.0f) return spikes;

        std::mt19937 rng(seed);
        std::uniform_real_distribution<float> uniform01(0.0f, 1.0f);
        std::uniform_real_distribution<float> azDist(0.0f, 359.9f);
        std::uniform_real_distribution<float> ampJitter(0.6f, 1.4f);
        std::uniform_real_distribution<float> phaseDist(0.0f, TWO_PI_F);
        std::uniform_int_distribution<int> typeDist(0, 2);

        const float durationSec = static_cast<float>(numSamples) / fs;
        std::exponential_distribution<float> expDist(p_.spikeRate);

        float t = 0.0f;
        while (t < durationSec)
        {
            t += expDist(rng);
            if (t >= durationSec) break;

            Spike s;
            s.sampleIndex = static_cast<int>(t * fs);
            s.azimuthDeg = (uniform01(rng) < p_.spikeSpatialFrac) ? azDist(rng) : -1.0f;
            s.type = typeDist(rng);
            s.amplitude = p_.spikeAmp * ampJitter(rng);
            s.phase = phaseDist(rng);

            if (s.type == 0)
            {
                s.tauSec = (p_.dropTauMinMs + uniform01(rng) *
                    (p_.dropTauMaxMs - p_.dropTauMinMs)) * 0.001f;
                s.f0 = s.f1 = 0.0f;
            }
            else if (s.type == 1)
            {
                s.tauSec = (p_.bubbleTauMinMs + uniform01(rng) *
                    (p_.bubbleTauMaxMs - p_.bubbleTauMinMs)) * 0.001f;
                s.f0 = p_.bubbleFreqStartMin + uniform01(rng) *
                    (p_.bubbleFreqStartMax - p_.bubbleFreqStartMin);
                s.f1 = p_.bubbleFreqEndMin + uniform01(rng) *
                    (p_.bubbleFreqEndMax - p_.bubbleFreqEndMin);
            }
            else
            {
                s.tauSec = (p_.clickTauMinMs + uniform01(rng) *
                    (p_.clickTauMaxMs - p_.clickTauMinMs)) * 0.001f;
                s.f0 = s.f1 = 0.0f;
            }
            spikes.push_back(s);
        }
        return spikes;
    }

    // ============================================================
    //  Генерация волн: регулярный поток с джиттером.
    //
    //  Было: exponential_distribution(waveRate) — интервалы
    //        экспоненциальные, первая волна может уехать далеко.
    //
    //  Стало: средний период = 1/waveRate, ±40% джиттера,
    //         первая волна в t=0.
    // ============================================================
    std::vector<OceanNoise::Wave>
    OceanNoise::generateWaves(int numSamples, float fs, uint32_t seed)
    {
        std::vector<Wave> waves;
        if (p_.waveRate <= 0.0f || p_.waveAmp <= 0.0f) return waves;

        std::mt19937 rng(seed);
        std::uniform_real_distribution<float> uniform01(0.0f, 1.0f);
        std::uniform_real_distribution<float> azDist(0.0f, 359.9f);
        std::uniform_real_distribution<float> phaseDist(0.0f, TWO_PI_F);
        std::uniform_real_distribution<float> ampJitter(0.7f, 1.3f);

        const float durationSec = static_cast<float>(numSamples) / fs;
        const float meanPeriod = 1.0f / p_.waveRate;
        const float jitterRange = 0.4f * meanPeriod;
        std::uniform_real_distribution<float> jitterDist(-jitterRange, +jitterRange);

        // Первая волна — ровно в t=0.
        float t = 0.0f;
        while (t < durationSec)
        {
            Wave w;
            w.sampleIndex = static_cast<int>(t * fs);
            if (w.sampleIndex >= numSamples) break;

            // Длительность волны — из диапазона.
            const float durSec = p_.waveDurationMinS + uniform01(rng) *
                (p_.waveDurationMaxS - p_.waveDurationMinS);
            w.lengthSamples = static_cast<int>(durSec * fs);

            // Обрезаем по концу буфера.
            if (w.sampleIndex + w.lengthSamples > numSamples)
            {
                w.lengthSamples = numSamples - w.sampleIndex;
            }
            if (w.lengthSamples <= 0) break;

            w.azimuthDeg = (uniform01(rng) < p_.waveSpatialFrac) ? azDist(rng) : -1.0f;
            w.amplitude = p_.waveAmp * ampJitter(rng);

            // --- θ-модуляция: азимут «плывёт» внутри волны ---
            // Размах качания — 30..60 градусов, случайно на каждую волну.
            w.azSweepDeg = 30.0f + uniform01(rng) * 30.0f;
            // Частота качания — 0.1..0.25 Гц (период 4..10 секунд).
            w.azSweepHz = 0.10f + uniform01(rng) * 0.15f;
            // Начальная фаза качания — случайна.
            w.azSweepPhase = phaseDist(rng);

            waves.push_back(w);

            // Следующая волна — через meanPeriod ± джиттер.
            t += meanPeriod + jitterDist(rng);
            if (t < 0.0f) t = 0.0f; // защита от ухода в минус
        }
        return waves;
    }

    // ============================================================
    //  addTo — БЕЗ ИЗМЕНЕНИЙ в логике, но вызываем новый renderWave.
    // ============================================================
    void OceanNoise::addTo(std::array<std::vector<float>, NUM_CHANNELS>& out,
                           int numSamples,
                           float fs,
                           const UcaGeometry& uca,
                           float soundSpeed)
    {
        // ---- 2. Всплески + эхо ----
        std::vector<Spike> spikes = generateSpikes(numSamples, fs, p_.seed + 7u);

        std::mt19937 rngRender(p_.seed + 21u);
        std::mt19937 rngEcho(p_.seed + 99u);
        std::uniform_real_distribution<float> echoDelayDist(p_.echoDelayMinMs, p_.echoDelayMaxMs);
        std::uniform_real_distribution<float> echoGainDist(p_.echoGainMin, p_.echoGainMax);
        std::uniform_real_distribution<float> echoAzSign(-1.0f, 1.0f);
        std::uniform_real_distribution<float> echoAzMag(10.0f, p_.echoAzSpreadDeg);

        for (const auto& s : spikes)
        {
            std::vector<float> mono(numSamples, 0.0f);
            renderSpike(mono, numSamples, fs, s.sampleIndex, s, rngRender);

            if (s.azimuthDeg >= 0.0f)
            {
                const auto dt = uca.arrivalOffsets(s.azimuthDeg, soundSpeed);
                for (int ch = 0; ch < NUM_CHANNELS; ++ch)
                {
                    const float dS = dt[ch] * fs;
                    for (int n = 0; n < numSamples; ++n)
                    {
                        const float src = static_cast<float>(n) - dS;
                        if (src >= 0.0f && src < static_cast<float>(numSamples - 1))
                        {
                            const int i0 = static_cast<int>(src);
                            const float fr = src - static_cast<float>(i0);
                            out[ch][n] += mono[i0] + fr * (mono[i0 + 1] - mono[i0]);
                        }
                    }
                }
            }
            else
            {
                const float perCh = 1.0f / NUM_CHANNELS;
                for (int ch = 0; ch < NUM_CHANNELS; ++ch)
                    for (int n = 0; n < numSamples; ++n) out[ch][n] += mono[n] * perCh;
            }

            if (p_.echoCount <= 0 || s.azimuthDeg < 0.0f) continue;
            for (int e = 0; e < p_.echoCount; ++e)
            {
                const int delayS = static_cast<int>(echoDelayDist(rngEcho) * 0.001f * fs);
                const int echoStart = s.sampleIndex + delayS;
                if (echoStart >= numSamples) continue;

                const float sign = (echoAzSign(rngEcho) >= 0.0f) ? 1.0f : -1.0f;
                float echoAz = s.azimuthDeg + sign * echoAzMag(rngEcho);
                while (echoAz < 0.0f) echoAz += 360.0f;
                while (echoAz >= 360.0f) echoAz -= 360.0f;
                const float echoGain = echoGainDist(rngEcho);

                Spike echo = s;
                echo.amplitude *= echoGain;

                std::vector<float> echoMono(numSamples, 0.0f);
                renderSpike(echoMono, numSamples, fs, echoStart, echo, rngRender);

                const auto dt = uca.arrivalOffsets(echoAz, soundSpeed);
                for (int ch = 0; ch < NUM_CHANNELS; ++ch)
                {
                    const float dch = dt[ch] * fs;
                    for (int n = 0; n < numSamples; ++n)
                    {
                        const float src = static_cast<float>(n) - dch;
                        if (src >= 0.0f && src < static_cast<float>(numSamples - 1))
                        {
                            const int i0 = static_cast<int>(src);
                            const float fr = src - static_cast<float>(i0);
                            out[ch][n] += echoMono[i0] + fr * (echoMono[i0 + 1] - echoMono[i0]);
                        }
                    }
                }
            }
        }

        // ---- 3. Волны ----
        std::vector<Wave> waves = generateWaves(numSamples, fs, p_.seed + 31u);
        std::mt19937 rngWave(p_.seed + 41u);

        // Огибающую волн считаем в ОТДЕЛЬНОМ буфере, а не из out.
        // Причина: в out уже могут быть спайки (блок 2) — они испортят
        // огибающую, и модуляция фона будет работать неправильно.
        std::vector<float> waveEnvelope(numSamples, 0.0f);

        // Отдельный временный буфер для каждой волны.
        std::array<std::vector<float>, NUM_CHANNELS> waveBuf;

        for (const auto& w : waves)
        {
            // Обнуляем временный буфер.
            for (int ch = 0; ch < NUM_CHANNELS; ++ch)
                waveBuf[ch].assign(numSamples, 0.0f);

            // Рисуем волну во временный буфер.
            renderWave(waveBuf, numSamples, fs, w, p_, uca, soundSpeed, rngWave);

            // Суммируем во временный буфер → в out.
            // И параллельно накапливаем огибающую.
            for (int n = 0; n < numSamples; ++n)
            {
                float s = 0.0f;
                for (int ch = 0; ch < NUM_CHANNELS; ++ch)
                {
                    const float v = waveBuf[ch][n];
                    out[ch][n] += v;
                    s += v * v;
                }
                waveEnvelope[n] += std::sqrt(s / NUM_CHANNELS);
            }
        }

        // ---- 3b. Фон, модулированный огибающей волн ----
        // Сглаживаем огибающую, чтобы не было резких скачков.
        // Однополюсный ФНЧ с медленной постоянной.
        {
            const float smoothCoef = 0.9995f;
            float smoothState = 0.0f;
            for (int n = 0; n < numSamples; ++n)
            {
                smoothState = smoothCoef * smoothState +
                    (1.0f - smoothCoef) * waveEnvelope[n];
                waveEnvelope[n] = smoothState;
            }

            // Нормируем огибающую в диапазон [0, 1].
            float maxEnv = 0.0f;
            for (int n = 0; n < numSamples; ++n)
                if (waveEnvelope[n] > maxEnv) maxEnv = waveEnvelope[n];
            if (maxEnv > 1e-6f)
                for (int n = 0; n < numSamples; ++n)
                    waveEnvelope[n] /= maxEnv;
            else
                std::fill(waveEnvelope.begin(), waveEnvelope.end(), 1.0f);

            // Теперь фон: базовый уровень × (1 - modulation + modulation*env).
            // modulation = 0 → фон ровный (множитель = 1).
            // modulation = 1 → фон строго следует за огибающей.
            const float m = std::clamp(p_.bgModulation, 0.0f, 1.0f);

            for (int ch = 0; ch < NUM_CHANNELS; ++ch)
            {
                std::vector<float> bg;
                generateBackgroundChannel(bg, numSamples, fs, p_.seed + 101u * (ch + 1));
                for (int n = 0; n < numSamples; ++n)
                {
                    const float env = waveEnvelope[n];
                    const float gain = (1.0f - m) + m * env;
                    out[ch][n] += bg[n] * gain;
                }
            }
        }

        // ---- 4. Модуляция (выключена) ----
        std::vector<float> mod = generateModulation(numSamples, fs, p_.seed + 13u);
        for (int ch = 0; ch < NUM_CHANNELS; ++ch)
            for (int n = 0; n < numSamples; ++n) out[ch][n] *= mod[n];

        // ---- 5. Санитайзер: NaN и inf → 0 ----
        for (int ch = 0; ch < NUM_CHANNELS; ++ch)
        {
            for (int n = 0; n < numSamples; ++n)
            {
                if (!std::isfinite(out[ch][n]))
                {
                    out[ch][n] = 0.0f;
                }
            }
        }
    }

// Линейная интерполяция: значение сигнала в точке t (дробный индекс).
// idx может быть отрицательным или >= numSamples — возвращаем 0.
static inline float sampleLinear(const std::vector<float>& x,
                                 float idx,
                                 int numSamples)
{
    if (idx < 0.0f || idx >= static_cast<float>(numSamples - 1)) {
        // За границей — нули (сигнал вне окна)
        return 0.0f;
    }
    const int i0 = static_cast<int>(idx);
    const float frac = idx - static_cast<float>(i0);
    const float s0 = x[i0];
    const float s1 = x[i0 + 1];
    return s0 + frac * (s1 - s0);
}

void Propagation::propagate(const std::vector<float>& monoSource,
                            int numSamples,
                            float fs,
                            const UcaGeometry& uca,
                            std::array<std::vector<float>, NUM_CHANNELS>& out)
{
    const int N = uca.numElements();
    if (N != NUM_CHANNELS) {
        // На всякий случай — наша UCA должна быть 8-элементной
        // (в будущем можно будет обобщить).
        return;
    }

    // 1. Задержки (генерационное соглашение — arrivalOffsets).
    const auto dt = uca.arrivalOffsets(p_.azimuthDeg, p_.soundSpeed);

    // 2. Уровень шума приёмника (относительно RMS сигнала).
    float rmsSignal = 0.0f;
    for (int i = 0; i < numSamples; ++i) {
        rmsSignal += monoSource[i] * monoSource[i];
    }
    rmsSignal = std::sqrt(rmsSignal / std::max(1, numSamples));

    float noiseRms = 0.0f;
    if (p_.receiverSnrDb > 100.0f) {
        noiseRms = 0.0f;
    } else {
        noiseRms = rmsSignal * std::pow(10.0f, -p_.receiverSnrDb / 20.0f);
    }

    // 3. Раскладка по каналам.
    std::mt19937 rng(p_.noiseSeed);
    std::normal_distribution<float> gauss(0.0f, 1.0f);

    for (int ch = 0; ch < N; ++ch) {
        out[ch].assign(numSamples, 0.0f);

        // Задержка в отсчётах: dt[ch] секунд → dt[ch]*fs отсчётов.
        const float delaySamples = dt[ch] * fs;

        for (int n = 0; n < numSamples; ++n) {
            // s_i(t_n) = s_source(t_n - dt_i)
            //           = monoSource[n - delaySamples]
            const float srcIdx = static_cast<float>(n) - delaySamples;
            const float s = sampleLinear(monoSource, srcIdx, numSamples);

            // Шум приёмника (независимый на каждом канале)
            const float noise = gauss(rng) * noiseRms;

            out[ch][n] = s + noise;
        }
    }
}

        void propagate(const std::vector<float>& monoSource,
                       int numSamples,
                       float fs,
                       const UcaGeometry& uca,
                       std::array<std::vector<float>, NUM_CHANNELS>& out);

    private:
        Params p_;

    void Scene::generate(std::array<std::vector<float>, NUM_CHANNELS>& out,
                         int numSamples,
                         float fs,
                         const UcaGeometry& uca)
    {
        // Проверка: минимум одна компонента.
        if (!hasAnyComponent())
        {
            throw std::runtime_error("Scene::generate: no components enabled");
        }

        // 1. Суммировать моно-сигналы всех пространственных источников.
        const bool hasSpatial = !sources_.empty();
        std::vector<float> mono(numSamples, 0.0f);

        if (hasSpatial)
        {
            std::vector<float> tmp(numSamples, 0.0f);
            for (auto& src : sources_)
            {
                // Обнуляем tmp ПЕРЕД генерацией каждого источника.
                // SourceModel::generate() ЗАПИСЫВАЕТ значения,
                // а не добавляет их, поэтому старые данные нужно стереть.
                std::fill(tmp.begin(), tmp.end(), 0.0f);

                src->reset();
                src->generate(tmp.data(), numSamples, fs, 0);
                for (int n = 0; n < numSamples; ++n)
                {
                    mono[n] += tmp[n];
                }
            }
        }

        // 2. Propagation → 8 каналов (или тишина).
        if (hasSpatial)
        {
            Propagation::Params pp;
            pp.azimuthDeg = params_.azimuthDeg;
            pp.soundSpeed = params_.soundSpeed;
            pp.receiverSnrDb = params_.receiverSnrDb;
            pp.noiseSeed = params_.noiseSeed;

            Propagation prop(pp);
            prop.propagate(mono, numSamples, fs, uca, out);
        }
        else
        {
            for (int ch = 0; ch < NUM_CHANNELS; ++ch)
            {
                out[ch].assign(numSamples, 0.0f);
            }
        }

        // 3. Шум океана.
        if (params_.oceanEnabled && params_.oceanParams.level > 0.0f)
        {
            OceanNoise ocean(params_.oceanParams);
            ocean.addTo(out, numSamples, fs, uca, params_.soundSpeed);
        }

        // ---- 4. Нормализация: гарантируем, что пик не превышает 0.95 ----
        // Это защищает от клиппинга при записи в WAV. Не меняет форму
        // сигнала — только уровень. Для каждого файла независимо.
        {
            float maxAbs = 0.0f;
            for (int ch = 0; ch < NUM_CHANNELS; ++ch)
            {
                for (int n = 0; n < numSamples; ++n)
                {
                    const float a = std::fabs(out[ch][n]);
                    if (a > maxAbs) maxAbs = a;
                }
            }

            if (maxAbs > 0.95f)
            {
                const float gain = 0.95f / maxAbs;
                for (int ch = 0; ch < NUM_CHANNELS; ++ch)
                {
                    for (int n = 0; n < numSamples; ++n)
                    {
                        out[ch][n] *= gain;
                    }
                }
            }
        }
    }
} // namespace hydro

    void PropellerSubmarine::regeneratePhases()
    {
        const int K = std::max(1, p_.numHarmonics);
        harmonicPhases_.resize(K);
        harmonicAmplitudes_.resize(K);

        std::uniform_real_distribution<float> phaseDist(0.0f, TWO_PI_F);
        for (int k = 0; k < K; ++k)
        {
            harmonicPhases_[k] = phaseDist(rng_);
            // Убывание: A_k = 1 / (k+1)
            harmonicAmplitudes_[k] = 1.0f / static_cast<float>(k + 1);
        }
    }

    void PropellerSubmarine::generate(float* out,
                                      int numSamples,
                                      float fs,
                                      int64_t sampleOffset)
    {
        const float f_blade = p_.bladeFreqHz();
        const float f_mod = p_.modulationHz;
        const float m = p_.modulationDepth;
        const float invFs = 1.0f / fs;

        // --- 1. Вычисляем амплитуду белого шума из SNR ---
        float P_sig = 0.0f;
        for (float A : harmonicAmplitudes_) P_sig += 0.5f * A * A;
        P_sig *= (1.0f + 0.5f * m * m);

        float P_noise = 0.0f;
        if (p_.snrDb > 100.0f)
        {
            P_noise = 0.0f;
        }
        else if (p_.snrDb < -40.0f)
        {
            P_noise = P_sig * 1e4f;
        }
        else
        {
            P_noise = P_sig / std::pow(10.0f, p_.snrDb / 10.0f);
        }
        const float noiseAmp = std::sqrt(P_noise);

        // --- 2. Кавитационный шум ---
        // Белый шум, модулированный медленной огибающей.
        // Простой белый, без фильтрации — как в эталонной версии.
        const float cavAmp = p_.cavitation;

        // --- 3. Генерация отсчётов ---
        for (int n = 0; n < numSamples; ++n)
        {
            const int64_t globalN = sampleOffset + n;
            const float t = static_cast<float>(globalN) * invFs;

            // 3a. Сумма гармоник с модуляцией
            float s = 0.0f;
            for (size_t k = 0; k < harmonicAmplitudes_.size(); ++k)
            {
                const float fk = f_blade * static_cast<float>(k + 1);
                const float phase = TWO_PI_F * fk * t + harmonicPhases_[k];
                s += harmonicAmplitudes_[k] * std::sin(phase);
            }

            // Амплитудная модуляция
            const float modFactor = 1.0f + m * std::sin(TWO_PI_F * f_mod * t);
            s *= modFactor;

            // 3b. Кавитационный шум — чистый белый
            const float cav = gauss_(rng_) * cavAmp * 0.5f;

            // 3c. Белый шум приёмника
            const float wgn = gauss_(rng_) * noiseAmp;

            out[n] = s + cav + wgn;
        }
    }

    void EngineSubmarine::regeneratePhases()
    {

        harmonicPhases_.resize(K);
        harmonicAmplitudes_.resize(K);

        std::uniform_real_distribution<float> phaseDist(0.0f, TWO_PI_F);

    void EngineSubmarine::generate(float* out,
                                   int numSamples,
                                   float fs,
                                   int64_t sampleOffset)
    {
        const float f_engine = p_.engineFreqHz();
        const float invFs = 1.0f / fs;

        // --- 1. Средняя мощность сигнала (для расчёта SNR) ---
        float P_sig = 0.0f;
        for (float A : harmonicAmplitudes_) P_sig += 0.5f * A * A;
        for (int r = 0; r < p_.numResonances && r < 3; ++r)
        {
            P_sig += 0.5f * RES_AMPS[r] * RES_AMPS[r];
        }

        // --- 2. Шум по SNR ---
        float P_noise = 0.0f;
        if (p_.snrDb > 100.0f)
        {
            P_noise = 0.0f;
        }
        else if (p_.snrDb < -40.0f)
        {
            P_noise = P_sig * 1e4f;
        }
        else
        {
            P_noise = P_sig / std::pow(10.0f, p_.snrDb / 10.0f);
        }
        const float noiseAmp = std::sqrt(P_noise);

        // --- 3. Генерация отсчётов ---
        for (int n = 0; n < numSamples; ++n)
        {
            const int64_t globalN = sampleOffset + n;
            const float t = static_cast<float>(globalN) * invFs;

            // 3a. Гармоники двигателя
            float s = 0.0f;
            for (size_t k = 0; k < harmonicAmplitudes_.size(); ++k)
            {
                const float fk = f_engine * static_cast<float>(k + 1);
                const float phase = TWO_PI_F * fk * t + harmonicPhases_[k];
                s += harmonicAmplitudes_[k] * std::sin(phase);
            }

            // 3b. Резонансы корпуса
            for (int r = 0; r < p_.numResonances && r < 3; ++r)
            {
                s += RES_AMPS[r] * std::sin(TWO_PI_F * RES_FREQS[r] * t);
            }

            // 3c. Механический широкополосный шум — чистый белый
            const float mech = gauss_(rng_) * p_.mechNoiseLevel * 0.3f;

            // 3d. Белый шум приёмника
            const float wgn = gauss_(rng_) * noiseAmp;

            out[n] = s + mech + wgn;
        }
    }

    // ------------------------------------------------------------
    //  Генерация событий: свисты
    //
    //  Свист — тональный сигнал с частотной модуляцией (ЧМ).
    //  Частота плавно меняется от f0 к f1 за время whistleDuration.
    //  Форма ЧМ — линейная (для простоты), но с добавлением
    //  небольшой случайности в направлении (вверх/вниз).
    // ------------------------------------------------------------
    std::vector<Dolphin::Whistle>
    Dolphin::generateWhistles(int numSamples, float fs, uint32_t seed)
    {
        std::vector<Whistle> whistles;
        if (p_.numWhistlesPerSec <= 0) return whistles;

        std::mt19937 rng(seed);
        std::uniform_real_distribution<float> uniform01(0.0f, 1.0f);
        std::uniform_real_distribution<float> phaseDist(0.0f, TWO_PI_F);

        const float durationSec = static_cast<float>(numSamples) / fs;

        // Средний период между свистами.
        const float meanPeriod = 1.0f / static_cast<float>(p_.numWhistlesPerSec);
        // Джиттер ±40%.
        const float jitterRange = 0.4f * meanPeriod;
        std::uniform_real_distribution<float> jitterDist(-jitterRange, +jitterRange);

        float t = 0.0f;
        while (t < durationSec)
        {
            Whistle w;

            w.sampleIndex = static_cast<int>(t * fs);
            if (w.sampleIndex >= numSamples) break;

            // Длительность свиста.
            const float durSec = p_.whistleDurationMin +
                uniform01(rng) * (p_.whistleDurationMax - p_.whistleDurationMin);
            w.lengthSamples = static_cast<int>(durSec * fs);
            if (w.sampleIndex + w.lengthSamples > numSamples)
            {
                w.lengthSamples = numSamples - w.sampleIndex;
            }
            if (w.lengthSamples <= 0) break;

            // Диапазон частот свиста.
            const float fLow = p_.whistleFreqMin;
            const float fHigh = p_.whistleFreqMax;
            const float fRange = fHigh - fLow;

            // Направление ЧМ: 0 — восходящий, 1 — нисходящий, 2 — U-образный.
            // Равновероятно.
            const int type = static_cast<int>(uniform01(rng) * 3.0f);
            if (type == 0)
            {
                // Восходящий: от fLow + немного к fHigh - немного.
                w.f0 = fLow + uniform01(rng) * fRange * 0.3f;
                w.f1 = fHigh - uniform01(rng) * fRange * 0.3f;
            }
            else if (type == 1)
            {
                // Нисходящий.
                w.f0 = fHigh - uniform01(rng) * fRange * 0.3f;
                w.f1 = fLow + uniform01(rng) * fRange * 0.3f;
            }
            else
            {
                // U-образный: сначала вниз, потом вверх — эмулируем
                // через начальную и конечную частоту, близкие друг к другу.
                // Реализуется как «сначала нисходящий, потом восходящий» —
                // но для простоты делаем нисходящий с возвратом.
                // В renderWhistle будем использовать нелинейную ЧМ.
                w.f0 = fHigh - uniform01(rng) * fRange * 0.2f;
                w.f1 = fHigh - uniform01(rng) * fRange * 0.2f;
            }

            w.phase = phaseDist(rng);

            whistles.push_back(w);

            // Следующий свист — через meanPeriod ± джиттер.
            t += meanPeriod + jitterDist(rng);
            if (t < 0.0f) t = 0.0f;
        }

        return whistles;
    }

    // ------------------------------------------------------------
    //  Рендер одного свиста в буфер.
    //
    //  Сигнал: sin(2*pi * f(t) * t + phase) * envelope(t),
    //  где f(t) меняется линейно от f0 к f1,
    //  envelope — плавное нарастание и спад (полкосинуса).
    //
    //  Гармоники: добавляем numHarmonics-1 обертонов с амплитудой
    //  1/k. Основная — 1.0.
    // ------------------------------------------------------------
    void Dolphin::renderWhistle(std::vector<float>& buffer,
                                int numSamples,
                                float fs,
                                const Whistle& w)
    {
        if (w.lengthSamples <= 0) return;

        const float invFs = 1.0f / fs;
        const float durSec = static_cast<float>(w.lengthSamples) * invFs;

        // Тип ЧМ: определяем по соотношению f0 и f1.
        // Если f0 < f1 — восходящий, если f0 > f1 — нисходящий,
        // если близки — U-образный.
        // Для U-образного сделаем f(t) = f0 - depth * sin(pi * t/dur).
        const float fDiff = w.f1 - w.f0;
        const bool isU = (std::fabs(fDiff) < 200.0f);

        const int K = std::max(1, p_.numHarmonics);

        for (int n = 0; n < w.lengthSamples; ++n)
        {
            const int idx = w.sampleIndex + n;
            if (idx < 0 || idx >= numSamples) continue;

            const float tLocal = static_cast<float>(n) * invFs;
            const float u = tLocal / durSec; // 0..1

            // Огибающая: плавный подъём и спад (полкосинус на краях).
            // Используем «окно Ханна»-подобное: 0.5*(1 - cos(2*pi*u)),
            // но для лучшей формы — комбинацию: rise 0.1, fall 0.1.
            float env = 1.0f;
            const float riseFrac = 0.1f;
            const float fallFrac = 0.1f;
            if (u < riseFrac)
            {
                env = 0.5f * (1.0f - std::cos(PI_F * u / riseFrac));
            }
            else if (u > 1.0f - fallFrac)
            {
                const float v = (1.0f - u) / fallFrac;
                env = 0.5f * (1.0f - std::cos(PI_F * v));
            }

            // Мгновенная частота.
            float fInst;
            if (isU)
            {
                // U-образная: f падает от f0 до f0 - depth, потом растёт.
                // depth = 0.4 * (fHigh - fLow), но берём из f0.
                const float depth = 0.4f * (p_.whistleFreqMax - p_.whistleFreqMin);
                fInst = w.f0 - depth * std::sin(PI_F * u);
            }
            else
            {
                // Линейная: от f0 к f1.
                fInst = w.f0 + fDiff * u;
            }

            // Основной тон + гармоники.
            float s = 0.0f;
            for (int k = 0; k < K; ++k)
            {
                const float fk = fInst * static_cast<float>(k + 1);
                // Отсечка выше Найквиста.
                if (fk >= fs * 0.5f) break;
                const float amp = 1.0f / static_cast<float>(k + 1);
                s += amp * std::sin(TWO_PI_F * fk * tLocal + w.phase);
            }

            buffer[idx] += s * env;
        }
    }

    // ------------------------------------------------------------
    //  Генерация событий: щелчки (эхолокационные, суррогатные).
    //
    //  Суррогатный щелчок — короткий импульс (50–150 мкс),
    //  заполненный широкополосным сигналом. Настоящие эхолокационные
    //  щелчки афалины — до 100+ кГц, при fs=22050 мы их не можем
    //  воспроизвести. Поэтому делаем «структурно похожий» сигнал:
    //  короткий, широкополосный, в диапазоне 1–8 кГц.
    //
    //  Щелчки идут сериями (click trains), 2–5 в серии.
    // ------------------------------------------------------------
    std::vector<Dolphin::Click>
    Dolphin::generateClicks(int numSamples, float fs, uint32_t seed)
    {
        std::vector<Click> clicks;
        if (p_.clickRate <= 0.0f || p_.clicksPerTrain <= 0) return clicks;

        std::mt19937 rng(seed);
        std::uniform_real_distribution<float> uniform01(0.0f, 1.0f);
        std::uniform_real_distribution<float> phaseDist(0.0f, TWO_PI_F);
        std::uniform_real_distribution<float> ampJitter(0.7f, 1.3f);

        const float durationSec = static_cast<float>(numSamples) / fs;
        const float meanPeriod = 1.0f / p_.clickRate;
        const float jitterRange = 0.4f * meanPeriod;
        std::uniform_real_distribution<float> jitterDist(-jitterRange, +jitterRange);

        float t = 0.0f;
        while (t < durationSec)
        {
            // Начало серии.
            const int trainStart = static_cast<int>(t * fs);
            if (trainStart >= numSamples) break;

            // Внутри серии — clicksPerTrain щелчков через ~50 мс.
            // Интервал между щелчками в серии — не параметр (фиксированный
            // реалистичный ~50 мс), потому что в реальности он у афалины
            // именно такой при эхолокации.
            const float intraTrainInterval = 0.05f; // 50 мс

            for (int k = 0; k < p_.clicksPerTrain; ++k)
            {
                Click c;
                c.sampleIndex = trainStart +
                    static_cast<int>(k * intraTrainInterval * fs);
                if (c.sampleIndex >= numSamples) break;

                c.amplitude = ampJitter(rng);
                c.phase = phaseDist(rng);

                clicks.push_back(c);
            }

            // Следующая серия.
            t += meanPeriod + jitterDist(rng);
            if (t < 0.0f) t = 0.0f;
        }

        return clicks;
    }

    // ------------------------------------------------------------
    //  Рендер одного щелчка.
    //
    //  Сигнал: короткий импульс с гауссовой огибающей,
    //  заполненный широкополосным сигналом.
    //
    //  Широкополосность эмулируем через сумму нескольких синусоид
    //  со случайными частотами в диапазоне clickFreqMin..Max,
    //  либо через белый шум × гауссова огибающая.
    //
    //  Второй способ проще и звучит похоже.
    // ------------------------------------------------------------
    void Dolphin::renderClick(std::vector<float>& buffer,
                              int numSamples,
                              float fs,
                              const Click& c)
    {
        const float durSec = p_.clickDurationUs * 1e-6f;
        const int len = static_cast<int>(durSec * fs) + 1;
        if (len <= 0) return;

        // Гауссова огибающая: exp(-(t/tau)^2), tau = durSec / 4.
        const float tau = durSec / 4.0f;
        const float invTau2 = 1.0f / (tau * tau);
        const float invFs = 1.0f / fs;

        // Генератор белого шума — локальный, чтобы не портить основной rng_.
        // Это важно: иначе порядок вызовов rng_ стал бы непредсказуемым
        // при изменении числа щелчков.
        std::mt19937 localRng(static_cast<uint32_t>(c.sampleIndex) ^ 0x9E3779B9u);
        std::normal_distribution<float> localGauss(0.0f, 1.0f);

        for (int n = 0; n < len; ++n)
        {
            const int idx = c.sampleIndex + n;
            if (idx < 0 || idx >= numSamples) continue;

            const float tLocal = static_cast<float>(n) * invFs;
            const float env = std::exp(-(tLocal * tLocal) * invTau2);

            // Белый шум как заполнение.
            const float white = localGauss(localRng);

            buffer[idx] += c.amplitude * env * white;
        }
    }

    // ------------------------------------------------------------
    //  Генерация событий: burst pulses (жужжание).
    //
    //  Burst — серия очень быстрых щелчков (интервал ~500 мкс),
    //  сливающихся в непрерывное жужжание. Длительность 0.3–1 сек.
    //
    //  Воспринимается как «трещотка» или «жужжание».
    // ------------------------------------------------------------
    std::vector<Dolphin::Burst>
    Dolphin::generateBursts(int numSamples, float fs, uint32_t seed)
    {
        std::vector<Burst> bursts;
        if (p_.burstRate <= 0.0f) return bursts;

        std::mt19937 rng(seed);
        std::uniform_real_distribution<float> uniform01(0.0f, 1.0f);
        std::uniform_real_distribution<float> phaseDist(0.0f, TWO_PI_F);

        const float durationSec = static_cast<float>(numSamples) / fs;
        const float meanPeriod = 1.0f / p_.burstRate;
        const float jitterRange = 0.4f * meanPeriod;
        std::uniform_real_distribution<float> jitterDist(-jitterRange, +jitterRange);

        float t = 0.0f;
        while (t < durationSec)
        {
            Burst b;
            b.sampleIndex = static_cast<int>(t * fs);
            if (b.sampleIndex >= numSamples) break;

            const float durSec = p_.burstDurationMin +
                uniform01(rng) * (p_.burstDurationMax - p_.burstDurationMin);
            b.lengthSamples = static_cast<int>(durSec * fs);
            if (b.sampleIndex + b.lengthSamples > numSamples)
            {
                b.lengthSamples = numSamples - b.sampleIndex;
            }
            if (b.lengthSamples <= 0) break;

            b.clickIntervalSamples = p_.burstClickIntervalUs * 1e-6f * fs;
            b.amplitude = 1.0f;
            b.phase = phaseDist(rng);

            bursts.push_back(b);

            t += meanPeriod + jitterDist(rng);
            if (t < 0.0f) t = 0.0f;
        }

        return bursts;
    }

    // ------------------------------------------------------------
    //  Рендер burst (жужжание).
    //
    //  Внутри burst генерируются короткие импульсы (щелчки)
    //  через clickIntervalSamples. Каждый импульс — короткий,
    //  заполнен белым шумом (как отдельный щелчок, но с меньшей
    //  длительностью).
    // ------------------------------------------------------------
    void Dolphin::renderBurst(std::vector<float>& buffer,
                              int numSamples,
                              float fs,
                              const Burst& b)
    {
        if (b.lengthSamples <= 0) return;

        const float durSec = p_.burstClickIntervalUs * 1e-6f;
        const int clickLen = static_cast<int>(durSec * fs * 0.5f) + 1;
        if (clickLen <= 0) return;

        const float tau = durSec / 4.0f;
        const float invTau2 = 1.0f / (tau * tau);
        const float invFs = 1.0f / fs;

        std::mt19937 localRng(static_cast<uint32_t>(b.sampleIndex) ^ 0x85EBCA6Bu);
        std::normal_distribution<float> localGauss(0.0f, 1.0f);

        // Идём по burst с шагом clickIntervalSamples.
        float pos = 0.0f;
        while (static_cast<int>(pos) < b.lengthSamples)
        {
            const int clickStart = b.sampleIndex + static_cast<int>(pos);
            for (int n = 0; n < clickLen; ++n)
            {
                const int idx = clickStart + n;
                if (idx < 0 || idx >= numSamples) break;
                // Ограничиваем внутри burst.
                if (idx >= b.sampleIndex + b.lengthSamples) break;

                const float tLocal = static_cast<float>(n) * invFs;
                const float env = std::exp(-(tLocal * tLocal) * invTau2);
                const float white = localGauss(localRng);

                buffer[idx] += b.amplitude * env * white;
            }

            pos += b.clickIntervalSamples;
        }
    }

    // ------------------------------------------------------------
    //  Основной generate для Dolphin.
    //
    //  Шаги:
    //    1. Генерируем события (свисты, щелчки, burst).
    //    2. Рендерим их в буфер.
    //    3. Добавляем белый шум по SNR.
    // ------------------------------------------------------------
    void Dolphin::generate(float* out,
                           int numSamples,
                           float fs,
                           int64_t sampleOffset)
    {
        // --- 1. Оценка средней мощности сигнала для расчёта SNR ---
        // Грубая оценка: сумма вклада всех трёх компонент.
        // Точная оценка была бы сложной (свисты, щелчки — события),
        // поэтому берём упрощённую: предполагаем, что средняя мощность
        // порядка единицы. Далее SNR настраиваем через порог.
        //
        // Поскольку события разрежены (2-3 свиста в секунду, редкие
        // щелчки), средняя мощность сильно ниже пиковой.
        // Это нормально — дельфин звучит как «импульсный» источник.

        // --- 2. Сброс RNG и генерация событий ---
        rng_.seed(p_.seed);

        // Отдельные под-seed'ы для каждой компоненты,
        // чтобы изменения одной не сдвигали другую.
        std::vector<Whistle> whistles = generateWhistles(numSamples, fs, p_.seed + 11u);
        std::vector<Click> clicks = generateClicks(numSamples, fs, p_.seed + 22u);
        std::vector<Burst> bursts = generateBursts(numSamples, fs, p_.seed + 33u);

        // --- 3. Буфер для накопления сигнала ---
        std::vector<float> buffer(numSamples, 0.0f);

        // --- 4. Рендер событий ---
        for (const auto& w : whistles) renderWhistle(buffer, numSamples, fs, w);
        for (const auto& c : clicks) renderClick(buffer, numSamples, fs, c);
        for (const auto& b : bursts) renderBurst(buffer, numSamples, fs, b);

        // --- 5. Расчёт RMS сигнала (для SNR) ---
        float sumSq = 0.0f;
        for (int n = 0; n < numSamples; ++n) sumSq += buffer[n] * buffer[n];
        const float rmsSignal = std::sqrt(sumSq / static_cast<float>(std::max(1, numSamples)));

        // --- 6. Амплитуда белого шума по SNR ---
        float noiseAmp = 0.0f;
        if (p_.snrDb <= 100.0f)
        {
            if (p_.snrDb < -40.0f)
            {
                noiseAmp = rmsSignal * 1e4f;
            }
            else
            {
                noiseAmp = rmsSignal * std::pow(10.0f, -p_.snrDb / 20.0f);
            }
        }

        // --- 7. Запись в out: сигнал + шум ---
        for (int n = 0; n < numSamples; ++n)
        {
            const float wgn = gauss_(rng_) * noiseAmp;
            out[n] = buffer[n] + wgn;
        }

        // sampleOffset в этой модели не используется: события
        // генерируются в локальных координатах от 0 до numSamples.
        // Это допустимо, потому что дельфин всегда генерируется
        // целиком на весь буфер сцены. Если понадобится генерация
        // по кускам — sampleOffset нужно будет учесть.
        (void)sampleOffset;
    }

    template <typename Channels>
    static bool writeWavFloat32(const std::string& path,
                                const Channels& channels,
                                int sampleRate)
    {
        if (channels.empty()) return false;
        const uint16_t numChannels = static_cast<uint16_t>(channels.size());
        const size_t numSamples = channels[0].size();
        for (const auto& ch : channels)
        {
            if (ch.size() != numSamples) return false;
        }

        std::ofstream f(path, std::ios::binary);
        if (!f) return false;

        const uint32_t dataBytes = static_cast<uint32_t>(numSamples * numChannels * 4);
        const uint32_t fmtChunkSize = 16;
        const uint16_t audioFormat = 3;
        const uint16_t bitsPerSample = 32;
        const uint32_t byteRate = sampleRate * numChannels * (bitsPerSample / 8);
        const uint16_t blockAlign = numChannels * (bitsPerSample / 8);
        const uint32_t riffSize = 4 + (8 + fmtChunkSize) + (8 + dataBytes);

        auto writeU32 = [&](uint32_t v) { f.write(reinterpret_cast<const char*>(&v), 4); };
        auto writeU16 = [&](uint16_t v) { f.write(reinterpret_cast<const char*>(&v), 2); };

        f.write("RIFF", 4);
        writeU32(riffSize);
        f.write("WAVE", 4);
        f.write("fmt ", 4);
        writeU32(fmtChunkSize);
        writeU16(audioFormat);
        writeU16(numChannels);
        writeU32(sampleRate);
        writeU32(byteRate);
        writeU16(blockAlign);
        writeU16(bitsPerSample);
        f.write("data", 4);
        writeU32(dataBytes);

        std::vector<float> frame(numChannels);
        for (size_t s = 0; s < numSamples; ++s)
        {
            for (uint16_t c = 0; c < numChannels; ++c)
            {
                frame[c] = channels[c][s];
            }
            f.write(reinterpret_cast<const char*>(frame.data()), numChannels * 4);
        }
        return true;
    }

    int DatasetGenerator::generateClass(
        const std::string& className,
        const std::function<std::unique_ptr<SourceModel>(std::mt19937&)>& sourceFactory,
        const std::function<std::string(const SourceModel*)>& jsonExtraFields,
        std::mt19937& rng,
        bool useOceanNoise) const
    {
        const fs::path classDir = fs::path(p_.outputDir) / className;
        fs::create_directories(classDir);

        const int totalSamples = static_cast<int>(p_.sampleRate) * p_.durationSec;
        int created = 0;

        // Общие диапазоны сцены.

        std::uniform_real_distribution<float> oceanLevelDist(ds::OCEAN_LEVEL_MIN, ds::OCEAN_LEVEL_MAX);

        for (int i = 1; i <= p_.filesPerClass; ++i)
        {
            // Создаём источник (для OceanNoise — nullptr).
            std::unique_ptr<SourceModel> source;
            const SourceModel* srcPtr = nullptr;
            if (!useOceanNoise)
            {
                source = sourceFactory(rng);
                if (!source) continue;
                srcPtr = source.get(); // сохраняем ДО передачи в Scene
            }

            SceneParams sp;
            if (useOceanNoise)
            {
                sp.azimuthDeg = -1.0f;
            }
            else if (p_.uniformAzimuth)
            {
                sp.azimuthDeg = (static_cast<float>(i - 1) / p_.filesPerClass) * 360.0f;
            }
            else
            {
                sp.azimuthDeg = p_.azimuthDeg;
            }
            sp.soundSpeed = SOUND_SPEED_WATER;
            sp.receiverSnrDb = p_.receiverSnrDb;
            sp.noiseSeed = static_cast<uint32_t>(rng());

            // Шум океана — включён для класса OceanNoise и для наложения на другие.
            fillOceanDefaults(sp.oceanParams);
            if (useOceanNoise)
            {
                sp.oceanEnabled = true;
                sp.oceanParams.level = 0.5f + oceanLevelDist(rng) * 0.5f; // 0.5–0.8
            }
            else
            {
                sp.oceanEnabled = true; // лёгкий фон для реализма
                sp.oceanParams.level = oceanLevelDist(rng);
            }
            sp.oceanParams.seed = static_cast<uint32_t>(rng());

            Scene scene;
            scene.setParams(sp);
            if (source) scene.addSource(std::move(source));

            std::array<std::vector<float>, NUM_CHANNELS> channels8;
            scene.generate(channels8, totalSamples, p_.sampleRate, uca_);

            const std::string wavName = makeFileName(i, ".wav");
            const fs::path wavPath = classDir / wavName;
            if (!writeWavFloat32(wavPath.string(), channels8,
                                 static_cast<int>(p_.sampleRate)))
            {
                std::cerr << "  [FAIL] " << wavPath << "\n";
                continue;
            }

            std::cout << "  [" << i << "/" << p_.filesPerClass << "] "
                << className << " #" << std::setw(4) << std::setfill('0') << i
                << "  az=" << std::setw(6) << std::setprecision(1)
                << std::fixed << sp.azimuthDeg
                << "°  recvSNR=" << std::setprecision(1)
                << sp.receiverSnrDb << " dB\n";

            ++created;
        }

        return created;
    }

        std::cout << "  files per class: " << p_.filesPerClass << "\n";
        std::cout << "  duration:        " << p_.durationSec << " s\n";
        std::cout << "  sample rate:     " << p_.sampleRate << " Hz\n";
        std::cout << "  output dir:      " << p_.outputDir << "\n\n";

        fs::create_directories(p_.outputDir);

        std::mt19937 rng(p_.masterSeed);

        // --- Диапазоны рандомизации ---
        std::uniform_int_distribution<int> bladesDist(ds::PROP_NUM_BLADES_MIN,
                                                      ds::PROP_NUM_BLADES_MAX);
        std::uniform_real_distribution<float> propRpmDist(ds::PROP_RPM_MIN,
                                                          ds::PROP_RPM_MAX);
        std::uniform_int_distribution<int> harmDist(ds::PROP_NUM_HARMONICS_MIN,
                                                    ds::PROP_NUM_HARMONICS_MAX);
        std::uniform_real_distribution<float> modDepthDist(ds::PROP_MOD_DEPTH_MIN,
                                                           ds::PROP_MOD_DEPTH_MAX);
        std::uniform_real_distribution<float> modHzDist(ds::PROP_MOD_HZ_MIN,
                                                        ds::PROP_MOD_HZ_MAX);
        std::uniform_real_distribution<float> cavDist(ds::PROP_CAVITATION_MIN,
                                                      ds::PROP_CAVITATION_MAX);
        std::uniform_real_distribution<float> propSnrDist(ds::PROP_SNR_DB_MIN,
                                                          ds::PROP_SNR_DB_MAX);

        std::uniform_real_distribution<float> engRpmDist(ds::ENG_RPM_MIN,
                                                         ds::ENG_RPM_MAX);
        std::uniform_int_distribution<int> engHarmDist(ds::ENG_NUM_HARMONICS_MIN,
                                                       ds::ENG_NUM_HARMONICS_MAX);
        std::uniform_int_distribution<int> engResDist(ds::ENG_NUM_RESONANCES_MIN,
                                                      ds::ENG_NUM_RESONANCES_MAX);
        std::uniform_real_distribution<float> engMechDist(ds::ENG_MECH_NOISE_MIN,
                                                          ds::ENG_MECH_NOISE_MAX);
        std::uniform_real_distribution<float> engSnrDist(ds::ENG_SNR_DB_MIN,
                                                         ds::ENG_SNR_DB_MAX);

        // --- Дельфин: диапазоны рандомизации ---
        std::uniform_int_distribution<int> dolphWhistlesDist(ds::DOLPHIN_WHISTLES_PER_SEC_MIN,
                                                             ds::DOLPHIN_WHISTLES_PER_SEC_MAX);
        std::uniform_real_distribution<float> dolphFreqMinDist(ds::DOLPHIN_WHISTLE_FREQ_MIN_MIN,
                                                               ds::DOLPHIN_WHISTLE_FREQ_MIN_MAX);
        std::uniform_real_distribution<float> dolphFreqMaxDist(ds::DOLPHIN_WHISTLE_FREQ_MAX_MIN,
                                                               ds::DOLPHIN_WHISTLE_FREQ_MAX_MAX);
        std::uniform_real_distribution<float> dolphClickRateDist(ds::DOLPHIN_CLICK_RATE_MIN,
                                                                 ds::DOLPHIN_CLICK_RATE_MAX);
        std::uniform_real_distribution<float> dolphBurstRateDist(ds::DOLPHIN_BURST_RATE_MIN,
                                                                 ds::DOLPHIN_BURST_RATE_MAX);
        std::uniform_real_distribution<float> dolphSnrDist(ds::DOLPHIN_SNR_DB_MIN,
                                                           ds::DOLPHIN_SNR_DB_MAX);

        int totalCreated = 0;

        // ---- PropellerSub ----
        if (sel_.propeller)
        {
            std::cout << "--- PropellerSub ---\n";
            totalCreated += generateClass(
                "PropellerSub",
                [&](std::mt19937& r) -> std::unique_ptr<SourceModel>
                {
                    PropellerParams pp;
                    pp.numBlades = bladesDist(r);
                    pp.rpm = propRpmDist(r);
                    pp.depthM = PROP_DEPTH_M;
                    pp.numHarmonics = harmDist(r);
                    pp.modulationDepth = modDepthDist(r);
                    pp.modulationHz = modHzDist(r);
                    pp.cavitation = cavDist(r);
                    pp.snrDb = propSnrDist(r);
                    pp.seed = static_cast<uint32_t>(r());
                    return std::make_unique<PropellerSubmarine>(pp);
                },
                [](const SourceModel* src) -> std::string
                {
                    const auto* p = dynamic_cast<const PropellerSubmarine*>(src);
                    if (!p) return "{}";
                    const auto& pp = p->params();
                    std::ostringstream os;
                    os << "{\n";
                    os << "    \"num_blades\": " << pp.numBlades << ",\n";
                    os << "    \"rpm\": " << std::fixed << std::setprecision(2) << pp.rpm << ",\n";
                    os << "    \"depth_m\": " << std::setprecision(1) << pp.depthM << ",\n";
                    os << "    \"cavitation\": " << std::setprecision(3) << pp.cavitation << ",\n";
                    os << "    \"snr_db\": " << std::setprecision(2) << pp.snrDb << ",\n";
                    os << "    \"seed\": " << pp.seed << "\n";
                    os << "  }";
                    return os.str();
                },
                rng, false);
        }

        // ---- EngineSub ----
        if (sel_.engine)
        {
            std::cout << "\n--- EngineSub ---\n";
            totalCreated += generateClass(
                "EngineSub",
                [&](std::mt19937& r) -> std::unique_ptr<SourceModel>
                {
                    EngineParams ep;
                    ep.rpm = engRpmDist(r);
                    ep.numHarmonics = engHarmDist(r);
                    ep.numResonances = engResDist(r);
                    ep.mechNoiseLevel = engMechDist(r);
                    ep.snrDb = engSnrDist(r);
                    ep.seed = static_cast<uint32_t>(r());
                    return std::make_unique<EngineSubmarine>(ep);
                },
                [](const SourceModel* src) -> std::string
                {
                    const auto* p = dynamic_cast<const EngineSubmarine*>(src);
                    if (!p) return "{}";
                    const auto& ep = p->params();
                    std::ostringstream os;
                    os << "{\n";
                    os << "    \"rpm\": " << std::fixed << std::setprecision(2) << ep.rpm << ",\n";
                    os << "    \"num_harmonics\": " << ep.numHarmonics << ",\n";
                    os << "    \"num_resonances\": " << ep.numResonances << ",\n";
                    os << "    \"mech_noise\": " << std::setprecision(3) << ep.mechNoiseLevel << ",\n";
                    os << "    \"snr_db\": " << std::setprecision(2) << ep.snrDb << ",\n";
                    os << "    \"seed\": " << ep.seed << "\n";
                    os << "  }";
                    return os.str();
                },
                rng, false);
        }

        // ---- Dolphin ----
        if (sel_.dolphin)
        {
            std::cout << "\n--- Dolphin ---\n";
            totalCreated += generateClass(
                "Dolphin",
                [&](std::mt19937& r) -> std::unique_ptr<SourceModel>
                {
                    DolphinParams dp;
                    dp.numWhistlesPerSec = dolphWhistlesDist(r);
                    dp.whistleDurationMin = DOLPHIN_WHISTLE_DUR_MIN;
                    dp.whistleDurationMax = DOLPHIN_WHISTLE_DUR_MAX;
                    dp.whistleFreqMin = dolphFreqMinDist(r);
                    dp.whistleFreqMax = dolphFreqMaxDist(r);
                    dp.whistleSweepDepth = DOLPHIN_WHISTLE_SWEEP;
                    dp.numHarmonics = DOLPHIN_NUM_HARMONICS;
                    dp.clickRate = dolphClickRateDist(r);
                    dp.clickDurationUs = DOLPHIN_CLICK_DUR_US;
                    dp.clickFreqMin = DOLPHIN_CLICK_FREQ_MIN;
                    dp.clickFreqMax = DOLPHIN_CLICK_FREQ_MAX;
                    dp.clicksPerTrain = DOLPHIN_CLICKS_PER_TRAIN;
                    dp.burstRate = dolphBurstRateDist(r);
                    dp.burstDurationMin = DOLPHIN_BURST_DUR_MIN;
                    dp.burstDurationMax = DOLPHIN_BURST_DUR_MAX;
                    dp.burstClickIntervalUs = DOLPHIN_BURST_CLICK_US;
                    dp.burstFreqMin = DOLPHIN_BURST_FREQ_MIN;
                    dp.burstFreqMax = DOLPHIN_BURST_FREQ_MAX;
                    dp.snrDb = dolphSnrDist(r);
                    dp.seed = static_cast<uint32_t>(r());
                    return std::make_unique<Dolphin>(dp);
                },
                [](const SourceModel* src) -> std::string
                {
                    const auto* p = dynamic_cast<const Dolphin*>(src);
                    if (!p) return "{}";
                    const auto& dp = p->params();
                    std::ostringstream os;
                    os << "{\n";
                    os << "    \"num_whistles_per_sec\": " << dp.numWhistlesPerSec << ",\n";
                    os << "    \"whistle_freq_min\": " << std::fixed << std::setprecision(1) << dp.whistleFreqMin <<
                        ",\n";
                    os << "    \"whistle_freq_max\": " << std::setprecision(1) << dp.whistleFreqMax << ",\n";
                    os << "    \"click_rate\": " << std::setprecision(3) << dp.clickRate << ",\n";
                    os << "    \"burst_rate\": " << std::setprecision(3) << dp.burstRate << ",\n";
                    os << "    \"snr_db\": " << std::setprecision(2) << dp.snrDb << ",\n";
                    os << "    \"seed\": " << dp.seed << "\n";
                    os << "  }";
                    return os.str();
                },
                rng, false);
        }

        // ---- OceanNoise ----
        if (sel_.ocean)
        {
            std::cout << "\n--- OceanNoise ---\n";
            totalCreated += generateClass(
                "OceanNoise",
                [](std::mt19937&) -> std::unique_ptr<SourceModel> { return nullptr; },
                [](const SourceModel*) -> std::string
                {
                    return "{\n    \"_note\": \"ocean noise only, no active source\"\n  }";
                },
                rng, true);
        }

        // ---- Combined — сумма всех выбранных компонент ----
        // Создаётся, только если выбрано 2+ компоненты.
        int selectedCount = (sel_.propeller ? 1 : 0)
            + (sel_.engine ? 1 : 0)
            + (sel_.dolphin ? 1 : 0)
            + (sel_.ocean ? 1 : 0);
        if (selectedCount >= 2)
        {
            std::cout << "\n--- Combined ---\n";
            totalCreated += generateCombinedClass(rng);
        }

        // ---- Сводный labels.json ----
        writeLabelsFile();

        // ---- _schema.json ----
        writeSchemaFile();

        std::cout << "\nTotal created: " << totalCreated << " WAV files.\n";
        std::cout << "Output: " << p_.outputDir << "\n";
        return totalCreated;
    }

    // ------------------------------------------------------------
    //  Сводный файл меток labels.json.
    // ------------------------------------------------------------
    void DatasetGenerator::writeLabelsFile() const
    {
        const fs::path path = fs::path(p_.outputDir) / "labels.json";
        std::ofstream f(path);
        if (!f) return;

        f << "{\n";
        f << "  \"version\": \"1.0\",\n";
        f << "  \"dataset_dir\": \"" << p_.outputDir << "\",\n";
        f << "  \"files_per_class\": " << p_.filesPerClass << ",\n";
        f << "  \"sample_rate\": " << static_cast<int>(p_.sampleRate) << ",\n";
        f << "  \"duration_sec\": " << p_.durationSec << ",\n";
        f << "  \"receiver_snr_db\": " << std::fixed << std::setprecision(2)
            << p_.receiverSnrDb << ",\n";
        f << "  \"files\": [\n";

        bool first = true;
        auto scanClass = [&](const std::string& className, bool isOmni)
        {
            const fs::path classDir = fs::path(p_.outputDir) / className;
            if (!fs::exists(classDir)) return;

            for (int i = 1; i <= p_.filesPerClass; ++i)
            {
                const fs::path wavPath = classDir / makeFileName(i, ".wav");
                if (!fs::exists(wavPath)) continue;

                float az;
                if (isOmni)
                {
                    az = -1.0f;
                }
                else if (p_.uniformAzimuth)
                {
                    az = (static_cast<float>(i - 1) / p_.filesPerClass) * 360.0f;
                }
                else
                {
                    az = p_.azimuthDeg;
                }

                if (!first) f << ",\n";
                first = false;

                f << "    { \"file\": \"" << className << "/" << makeFileName(i, ".wav")
                    << "\", \"class\": \"" << className
                    << "\", \"azimuth_deg\": " << std::setprecision(2) << az
                    << ", \"receiver_snr_db\": " << std::setprecision(2) << p_.receiverSnrDb
                    << " }";
            }
        };

        if (sel_.propeller) scanClass("PropellerSub", false);
        if (sel_.engine) scanClass("EngineSub", false);
        if (sel_.dolphin) scanClass("Dolphin", false);
        if (sel_.ocean) scanClass("OceanNoise", true);

        // Combined — только если выбрано 2+ компоненты.
        int selectedCount = (sel_.propeller ? 1 : 0)
            + (sel_.engine ? 1 : 0)
            + (sel_.dolphin ? 1 : 0)
            + (sel_.ocean ? 1 : 0);
        if (selectedCount >= 2) scanClass("Combined", false);

        f << "\n  ]\n";
        f << "}\n";
    }

    // ------------------------------------------------------------
    //  Combined — сумма всех ВЫБРАННЫХ компонент в одном файле.
    // ------------------------------------------------------------
    int DatasetGenerator::generateCombinedClass(std::mt19937& rng) const
    {
        const fs::path classDir = fs::path(p_.outputDir) / "Combined";
        fs::create_directories(classDir);

        const int totalSamples = static_cast<int>(p_.sampleRate) * p_.durationSec;
        int created = 0;

        // Диапазоны рандомизации (те же, что и в generate()).
        std::uniform_int_distribution<int> bladesDist(ds::PROP_NUM_BLADES_MIN,
                                                      ds::PROP_NUM_BLADES_MAX);
        std::uniform_real_distribution<float> propRpmDist(ds::PROP_RPM_MIN,
                                                          ds::PROP_RPM_MAX);
        std::uniform_int_distribution<int> harmDist(ds::PROP_NUM_HARMONICS_MIN,
                                                    ds::PROP_NUM_HARMONICS_MAX);
        std::uniform_real_distribution<float> modDepthDist(ds::PROP_MOD_DEPTH_MIN,
                                                           ds::PROP_MOD_DEPTH_MAX);
        std::uniform_real_distribution<float> modHzDist(ds::PROP_MOD_HZ_MIN,
                                                        ds::PROP_MOD_HZ_MAX);
        std::uniform_real_distribution<float> cavDist(ds::PROP_CAVITATION_MIN,
                                                      ds::PROP_CAVITATION_MAX);
        std::uniform_real_distribution<float> propSnrDist(ds::PROP_SNR_DB_MIN,
                                                          ds::PROP_SNR_DB_MAX);

        std::uniform_real_distribution<float> engRpmDist(ds::ENG_RPM_MIN,
                                                         ds::ENG_RPM_MAX);
        std::uniform_int_distribution<int> engHarmDist(ds::ENG_NUM_HARMONICS_MIN,
                                                       ds::ENG_NUM_HARMONICS_MAX);
        std::uniform_int_distribution<int> engResDist(ds::ENG_NUM_RESONANCES_MIN,
                                                      ds::ENG_NUM_RESONANCES_MAX);
        std::uniform_real_distribution<float> engMechDist(ds::ENG_MECH_NOISE_MIN,
                                                          ds::ENG_MECH_NOISE_MAX);
        std::uniform_real_distribution<float> engSnrDist(ds::ENG_SNR_DB_MIN,
                                                         ds::ENG_SNR_DB_MAX);

        std::uniform_int_distribution<int> dolphWhistlesDist(ds::DOLPHIN_WHISTLES_PER_SEC_MIN,
                                                             ds::DOLPHIN_WHISTLES_PER_SEC_MAX);
        std::uniform_real_distribution<float> dolphFreqMinDist(ds::DOLPHIN_WHISTLE_FREQ_MIN_MIN,
                                                               ds::DOLPHIN_WHISTLE_FREQ_MIN_MAX);
        std::uniform_real_distribution<float> dolphFreqMaxDist(ds::DOLPHIN_WHISTLE_FREQ_MAX_MIN,
                                                               ds::DOLPHIN_WHISTLE_FREQ_MAX_MAX);
        std::uniform_real_distribution<float> dolphClickRateDist(ds::DOLPHIN_CLICK_RATE_MIN,
                                                                 ds::DOLPHIN_CLICK_RATE_MAX);
        std::uniform_real_distribution<float> dolphBurstRateDist(ds::DOLPHIN_BURST_RATE_MIN,
                                                                 ds::DOLPHIN_BURST_RATE_MAX);
        std::uniform_real_distribution<float> dolphSnrDist(ds::DOLPHIN_SNR_DB_MIN,
                                                           ds::DOLPHIN_SNR_DB_MAX);

        std::uniform_real_distribution<float> oceanLevelDist(ds::OCEAN_LEVEL_MIN,
                                                             ds::OCEAN_LEVEL_MAX);

        for (int i = 1; i <= p_.filesPerClass; ++i)
        {
            SceneParams sp;
            if (p_.uniformAzimuth)
                sp.azimuthDeg = (static_cast<float>(i - 1) / p_.filesPerClass) * 360.0f;
            else
                sp.azimuthDeg = p_.azimuthDeg;

            sp.soundSpeed = SOUND_SPEED_WATER;
            sp.receiverSnrDb = p_.receiverSnrDb;
            sp.noiseSeed = rng();

            // Океан — включаем, если выбран.
            fillOceanDefaults(sp.oceanParams);
            sp.oceanEnabled = sel_.ocean;
            if (sel_.ocean)
            {
                sp.oceanParams.level = oceanLevelDist(rng);
            }
            sp.oceanParams.seed = rng();

            Scene scene;
            scene.setParams(sp);

            // Добавляем выбранные пространственные источники.
            if (sel_.propeller)
            {
                PropellerParams pp;
                pp.numBlades = bladesDist(rng);
                pp.rpm = propRpmDist(rng);
                pp.depthM = PROP_DEPTH_M;
                pp.numHarmonics = harmDist(rng);
                pp.modulationDepth = modDepthDist(rng);
                pp.modulationHz = modHzDist(rng);
                pp.cavitation = cavDist(rng);
                pp.snrDb = propSnrDist(rng);
                pp.seed = static_cast<uint32_t>(rng());
                scene.addSource(std::make_unique<PropellerSubmarine>(pp));
            }
            if (sel_.engine)
            {
                EngineParams ep;
                ep.rpm = engRpmDist(rng);
                ep.numHarmonics = engHarmDist(rng);
                ep.numResonances = engResDist(rng);
                ep.mechNoiseLevel = engMechDist(rng);
                ep.snrDb = engSnrDist(rng);
                ep.seed = static_cast<uint32_t>(rng());
                scene.addSource(std::make_unique<EngineSubmarine>(ep));
            }
            if (sel_.dolphin)
            {
                DolphinParams dp;
                dp.numWhistlesPerSec = dolphWhistlesDist(rng);
                dp.whistleDurationMin = DOLPHIN_WHISTLE_DUR_MIN;
                dp.whistleDurationMax = DOLPHIN_WHISTLE_DUR_MAX;
                dp.whistleFreqMin = dolphFreqMinDist(rng);
                dp.whistleFreqMax = dolphFreqMaxDist(rng);
                dp.whistleSweepDepth = DOLPHIN_WHISTLE_SWEEP;
                dp.numHarmonics = DOLPHIN_NUM_HARMONICS;
                dp.clickRate = dolphClickRateDist(rng);
                dp.clickDurationUs = DOLPHIN_CLICK_DUR_US;
                dp.clickFreqMin = DOLPHIN_CLICK_FREQ_MIN;
                dp.clickFreqMax = DOLPHIN_CLICK_FREQ_MAX;
                dp.clicksPerTrain = DOLPHIN_CLICKS_PER_TRAIN;
                dp.burstRate = dolphBurstRateDist(rng);
                dp.burstDurationMin = DOLPHIN_BURST_DUR_MIN;
                dp.burstDurationMax = DOLPHIN_BURST_DUR_MAX;
                dp.burstClickIntervalUs = DOLPHIN_BURST_CLICK_US;
                dp.burstFreqMin = DOLPHIN_BURST_FREQ_MIN;
                dp.burstFreqMax = DOLPHIN_BURST_FREQ_MAX;
                dp.snrDb = dolphSnrDist(rng);
                dp.seed = static_cast<uint32_t>(rng());
                scene.addSource(std::make_unique<Dolphin>(dp));
            }

            std::array<std::vector<float>, NUM_CHANNELS> ch8;
            scene.generate(ch8, totalSamples, p_.sampleRate, uca_);

            const std::string wavName = makeFileName(i, ".wav");
            const fs::path wavPath = classDir / wavName;
            if (!writeWavFloat32(wavPath.string(), ch8,
                                 static_cast<int>(p_.sampleRate)))
            {
                std::cerr << "  [FAIL] " << wavPath << "\n";
                continue;
            }

            std::cout << "  [" << i << "/" << p_.filesPerClass << "] "
                << "Combined #" << std::setw(4) << std::setfill('0') << i
                << "  az=" << std::setw(6) << std::setprecision(1)
                << std::fixed << sp.azimuthDeg
                << "°\n";

            ++created;
        }

        return created;
    }

    // ============================================================
    //  Генератор синтетического датасета.
    //
    //  Создаёт N сэмплов на каждый ВЫБРАННЫЙ класс:
    //    dataset/<ClassName>/<NNNN>.wav  — 8-канальный WAV, 32-bit float
    //    dataset/<ClassName>/<NNNN>.json — метаданные
    //
    //  Плюс один раз пишет dataset/_schema.json.
    //
    //  Директории относительные — считаются от рабочей директории запуска.
    // ============================================================

    struct DatasetParams
    {
        int filesPerClass = 10; // файлов на класс
        int durationSec = 2; // длительность сэмпла
        float sampleRate = 10000.0f; // fs
        std::string outputDir = "dataset"; // куда писать
        uint32_t masterSeed = 20260924; // seed рандомизации

        // --- Общие параметры сцены (одинаковые для всех файлов) ---
        float receiverSnrDb = 30.0f; // SNR приёмника

        // --- Азимут ---
        bool uniformAzimuth = true; // true = равномерно 0..359
        // false = фиксированный azimuthDeg
        float azimuthDeg = 45.0f; // используется если uniformAzimuth = false
    };

    // Что генерировать. Минимум один класс должен быть включён.
    struct DatasetSelection
    {
        bool propeller = true;
        bool engine = true;
        bool dolphin = true;
        bool ocean = true;
    };

    class DatasetGenerator
    {
    public:
        DatasetGenerator(const UcaGeometry& uca,
                         const DatasetParams& params,
                         const DatasetSelection& selection);

        // Сгенерировать выбранные классы.
        // Возвращает число успешно созданных WAV-файлов.
        int generate();

    private:
        const UcaGeometry& uca_;
        DatasetParams p_;
        DatasetSelection sel_;

        // --- Один класс ---
        // className       — имя класса ("PropellerSub", "EngineSub", "Dolphin", "OceanNoise")
        // sourceFactory   — лямбда, создающая источник (или nullptr для OceanNoise)
        // jsonExtraFields — функция, формирующая params для JSON
        // rng             — общий RNG
        // useOceanNoise   — если true, генерируется только шум океана (без сигнала)
        int generateClass(
            const std::string& className,
            const std::function<std::unique_ptr<SourceModel>(std::mt19937&)>& sourceFactory,
            const std::function<std::string(const SourceModel*)>& jsonExtraFields,
            std::mt19937& rng,
            bool useOceanNoise) const;

        void writeSchemaFile() const;
        void writeLabelsFile() const;
        int generateCombinedClass(std::mt19937& rng) const;
        static std::string makeFileName(int index, const std::string& ext);
    };
